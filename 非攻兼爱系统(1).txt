/// The unified error type for the 非攻兼爱推演服务平台.
///
/// Covers all failure domains: cache, inference, I/O, serialization,
/// storage, network, circuit breaker, rate limiting, timeouts, model
/// lookup, input validation, and resource quotas.
#[derive(Debug, Clone, Error)]
pub enum FeiMingError {
    #[error("cache error: {0}")]
    Cache(String),
    #[error("inference error: {0}")]
    Inference(String),
    #[error("io error: {0}")]
    Io(String),
    #[error("serialization error: {0}")]
    Serialization(String),
    #[error("storage error: {0}")]
    Storage(String),
    #[error("network error: {0}")]
    Network(String),
    #[error("circuit breaker open")]
    CircuitBreakerOpen,
    #[error("rate limited")]
    RateLimited,
    #[error("timeout")]
    Timeout,
    #[error("model not found: {0}")]
    ModelNotFound(MoQiHao),
    #[error("invalid input: {0}")]
    InvalidInput(String),
    #[error("quota exceeded")]
    QuotaExceeded,
    #[error("unknown error: {0}")]
    Unknown(String),
}

impl From<std::io::Error> for FeiMingError {
    fn from(err: std::io::Error) -> Self {
        FeiMingError::Io(err.to_string())
    }
}

impl From<serde_json::Error> for FeiMingError {
    fn from(err: serde_json::Error) -> Self {
        FeiMingError::Serialization(err.to_string())
    }
}

impl From<Box<bincode::ErrorKind>> for FeiMingError {
    fn from(err: Box<bincode::ErrorKind>) -> Self {
        FeiMingError::Serialization(err.to_string())
    }
}

impl From<anyhow::Error> for FeiMingError {
    fn from(err: anyhow::Error) -> Self {
        FeiMingError::Unknown(err.to_string())
    }
}

impl From<prometheus::Error> for FeiMingError {
    fn from(err: prometheus::Error) -> Self {
        FeiMingError::Unknown(err.to_string())
    }
}

/// A multi-dimensional array with a named type, shape, and raw byte payload.
///
/// Tensors are the fundamental data carrier for inference requests and
/// responses. The `data` field holds a contiguous flattened buffer whose
/// layout follows row-major (C) order.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub struct Tensor {
    /// Human-readable name of the tensor (e.g. "input_ids").
    pub name: String,
    /// Element data type.
    pub dtype: DType,
    /// Shape dimensions. An empty slice denotes a scalar.
    pub shape: Vec<i64>,
    /// Raw flattened byte data.
    pub data: Vec<u8>,
}

impl Tensor {
    /// Create a new `Tensor`.
    pub fn new(name: impl Into<String>, dtype: DType, shape: Vec<i64>, data: Vec<u8>) -> Self {
        Tensor {
            name: name.into(),
            dtype,
            shape,
            data,
        }
    }

    /// Return the number of elements in the tensor.
    ///
    /// Returns `1` for a scalar (empty shape), `0` if any dimension is
    /// non-positive, and the product of all dimensions otherwise.
    pub fn element_count(&self) -> usize {
        if self.shape.is_empty() {
            return 1;
        }
        let mut count: usize = 1;
        for &dim in &self.shape {
            if dim <= 0 {
                return 0;
            }
            count *= dim as usize;
        }
        count
    }

    /// Return the byte length of the raw data buffer.
    pub fn byte_size(&self) -> usize {
        self.data.len()
    }

    /// Validate that the tensor has exactly `R` dimensions.
    ///
    /// Returns `Ok(())` when `shape.len() == R`, otherwise returns
    /// `FeiMingError::InvalidInput`.
    pub fn check_rank<const R: usize>(&self) -> Result<()> {
        if self.shape.len() == R {
            Ok(())
        } else {
            Err(FeiMingError::InvalidInput(format!(
                "expected rank {}, got {}",
                R,
                self.shape.len()
            )))
        }
    }

    /// Reinterpret the raw data buffer as a slice of type `T`.
    ///
    /// Checks that `T` has non-zero size, that `data.len()` is divisible
    /// by `size_of::<T>()`, and that the data pointer is properly
    /// aligned for `T`. Returns `FeiMingError::InvalidInput` on mismatch.
    pub fn as_typed_slice<T: Copy>(&self) -> Result<&[T]> {
        let type_size = std::mem::size_of::<T>();
        if type_size == 0 {
            return Err(FeiMingError::InvalidInput(
                "zero-sized types are not supported".into(),
            ));
        }
        if self.data.len() % type_size != 0 {
            return Err(FeiMingError::InvalidInput(format!(
                "data length {} is not divisible by type size {}",
                self.data.len(),
                type_size
            )));
        }
        if self.data.as_ptr().align_offset(std::mem::align_of::<T>()) != 0 {
            return Err(FeiMingError::InvalidInput(
                "data buffer is not properly aligned for the requested type".into(),
            ));
        }
        let elem_count = self.data.len() / type_size;
        let slice = unsafe { std::slice::from_raw_parts(self.data.as_ptr().cast::<T>(), elem_count) };
        Ok(slice)
    }
}

/// A single cached inference result stored in one of the cache tiers.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MoYin {
    /// Composite cache key derived from `model_id` and `input_hash`.
    pub key: MoYue,
    /// Model that produced this result.
    pub model_id: MoQiHao,
    /// xxHash of the input tensors.
    pub input_hash: u64,
    /// Serialized output data (e.g. compressed protobuf or JSON).
    pub output_data: Vec<u8>,
    /// Unix timestamp (seconds) when the entry was created.
    pub created_at: u64,
    /// Unix timestamp (seconds) when the entry expires.
    pub expires_at: u64,
    /// Inference latency in microseconds (stored for analytics).
    pub inference_time_us: u64,
}

impl MoYin {
    /// Create a new `MoYin` with the current time as `created_at`
    /// and `expires_at = now + ttl_secs`.
    pub fn new(
        model_id: MoQiHao,
        input_hash: u64,
        output_data: Vec<u8>,
        ttl_secs: u64,
        inference_time_us: u64,
    ) -> Self {
        let key = Self::make_key(&model_id, input_hash);
        let now = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap_or_default()
            .as_secs();
        MoYin {
            key,
            model_id,
            input_hash,
            output_data,
            created_at: now,
            expires_at: now + ttl_secs,
            inference_time_us,
        }
    }

    /// Build a canonical cache key from a model id and input hash.
    pub fn make_key(model_id: &str, input_hash: u64) -> MoYue {
        format!("{}:{:016x}", model_id, input_hash)
    }

    /// Return `true` if the entry has passed its expiration time.
    pub fn is_expired(&self) -> bool {
        let now = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap_or_default()
            .as_secs();
        now >= self.expires_at
    }

    /// Return the remaining TTL in seconds, or `0` if already expired.
    pub fn remaining_ttl_secs(&self) -> u64 {
        let now = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap_or_default()
            .as_secs();
        self.expires_at.saturating_sub(now)
    }

    /// Serialize the entry to bytes using bincode.
    pub fn to_bytes(&self) -> Result<Vec<u8>> {
        bincode::serialize(self).map_err(FeiMingError::from)
    }

    /// Deserialize an entry from a bincode byte slice.
    pub fn from_bytes(data: &[u8]) -> Result<Self> {
        bincode::deserialize(data).map_err(FeiMingError::from)
    }

    /// Return the byte size of the stored output data.
    pub fn byte_size(&self) -> usize {
        self.output_data.len()
    }
}

/// Pluggable encoding strategy for [`MoYin`] serialization.
pub trait MoFaJie: Send + Sync + 'static {
    /// Encode a [`MoYin`] into a byte vector.
    fn encode(entry: &MoYin) -> Result<Vec<u8>>;
    /// Decode a [`MoYin`] from a byte slice.
    fn decode(data: &[u8]) -> Result<MoYin>;
}

/// JSON-based [`MoFaJie`] implementation.
pub struct MingWenFa;

impl MoFaJie for MingWenFa {
    fn encode(entry: &MoYin) -> Result<Vec<u8>> {
        let json_str = serde_json::to_string(entry)?;
        Ok(json_str.into_bytes())
    }

    fn decode(data: &[u8]) -> Result<MoYin> {
        let entry: MoYin = serde_json::from_slice(data)?;
        Ok(entry)
    }
}

/// Bincode-based [`MoFaJie`] implementation (default, faster).
pub struct MiFengFa;

impl MoFaJie for MiFengFa {
    fn encode(entry: &MoYin) -> Result<Vec<u8>> {
        let bytes = bincode::serialize(entry).map_err(FeiMingError::from)?;
        Ok(bytes)
    }

    fn decode(data: &[u8]) -> Result<MoYin> {
        let entry: MoYin = bincode::deserialize(data).map_err(FeiMingError::from)?;
        Ok(entry)
    }
}

/// A compact, unexpanded skill descriptor sent to the expansion service.
#[derive(Serialize, Deserialize, Debug, Clone)]
pub struct JiZhong {
    /// Unique identifier assigned at creation time.
    pub id: Uuid,
    /// Human-readable skill name.
    pub name: String,
    /// Opaque binary payload (e.g. compressed protobuf, raw bytes).
    pub payload: Vec<u8>,
    /// Optional free-form metadata string.
    pub meta: Option<String>,
    /// Optional kind/category tag.
    pub kind: Option<String>,
}

impl JiZhong {
    /// Create a new `JiZhong` with a fresh UUID.
    pub fn new(name: impl Into<String>, payload: Vec<u8>) -> Self {
        JiZhong {
            id: Uuid::new_v4(),
            name: name.into(),
            payload,
            meta: None,
            kind: None,
        }
    }
}

    pub id: Uuid,

    pub name: String,
    /// JSON value holding all expanded parameters.
    pub params: JsonValue,
    /// Optional runtime handle (e.g. a shared library path).
    pub runtime_handle: Option<String>,

    pub expanded_meta: Option<String>,
}

    /// Element data type of the output.

    pub shape: Vec<Option<usize>>,
}

/// Convert a [`pb::TensorProto`] into a [`Tensor`].
///
/// Returns [`FeiMingError::InvalidInput`] if the `dtype` field does not
/// correspond to a known [`DType`] variant.
pub fn tensor_from_proto(proto: &pb::TensorProto) -> Result<Tensor> {
    let dtype = match proto.dtype {
        0 => DType::F32,
        1 => DType::F64,
        2 => DType::I32,
        3 => DType::I64,
        4 => DType::U8,
        5 => DType::I8,
        6 => DType::Bool,
        other => {
            return Err(FeiMingError::InvalidInput(format!(
                "unknown dtype discriminant: {}",
                other
            )))
        }
    };
    Ok(Tensor {
        name: proto.name.clone(),
        dtype,
        shape: proto.shape.clone(),
        data: proto.data.clone(),
    })
}

/// Convert a [`Tensor`] into a [`pb::TensorProto`].
pub fn tensor_to_proto(tensor: &Tensor) -> pb::TensorProto {
    pb::TensorProto {
        name: tensor.name.clone(),
        dtype: tensor.dtype as i32,
        shape: tensor.shape.clone(),
        data: tensor.data.clone(),
    }
}

// ---------------------------------------------------------------------------
// hash_inputs
// ---------------------------------------------------------------------------

/// Compute a 64-bit xxHash over a slice of input tensors.
///
/// The hash covers every tensor's name, dtype discriminant, shape
/// dimensions (in little-endian), and raw data bytes. This produces a
/// stable cache key for identical input payloads.
pub fn hash_inputs(inputs: &[Tensor]) -> u64 {
    let mut hasher = Xxh3::new();
    for tensor in inputs {
        hasher.update(tensor.name.as_bytes());
        hasher.update(&[tensor.dtype as u8]);
        for &dim in &tensor.shape {
            hasher.update(&dim.to_le_bytes());
        }
        hasher.update(&tensor.data);
    }
    hasher.digest()
}

// ---------------------------------------------------------------------------
// bincode_encode / bincode_decode / zstd_compress / zstd_decompress
// ---------------------------------------------------------------------------

/// Serialize a value using bincode.
///
/// Returns [`FeiMingError::Serialization`] on failure.
pub fn bincode_encode<T: Serialize>(value: &T) -> Result<Vec<u8>> {
    bincode::serialize(value).map_err(|e| FeiMingError::Serialization(e.to_string()))
}

/// Deserialize a value using bincode.
///
/// Returns [`FeiMingError::Serialization`] on failure.
pub fn bincode_decode<T: for<'de> Deserialize<'de>>(data: &[u8]) -> Result<T> {
    bincode::deserialize(data).map_err(|e| FeiMingError::Serialization(e.to_string()))
}

/// Compress a byte slice using zstd with the store compression level.
///
/// Returns [`FeiMingError::Serialization`] (compression failure) on error.
pub fn zstd_compress(data: &[u8]) -> Result<Vec<u8>> {
    zstd::encode_all(data, ZSTD_YASUO_DENGJI)
        .map_err(|e| FeiMingError::Serialization(format!("zstd compress failed: {}", e)))
}

/// Decompress a zstd-compressed byte slice.
///
/// Returns [`FeiMingError::Serialization`] (decompression failure) on error.
pub fn zstd_decompress(data: &[u8]) -> Result<Vec<u8>> {
    zstd::decode_all(data)
        .map_err(|e| FeiMingError::Serialization(format!("zstd decompress failed: {}", e)))
}

// ---------------------------------------------------------------------------

/// Abstract interface for durable [`JiZhong`] storage.
///
/// All backends guarantee atomicity of individual `put` / `get` / `remove`
/// operations.  Implementors must be `Send + Sync + 'static` so they can
/// be held inside an `Arc<dyn JiZhongCang>` and used across Tokio tasks.
#[async_trait]
pub trait JiZhongCang: Send + Sync + 'static {
    /// Persist a [`JiZhong`] under the given logical key.
    ///
    /// Overwrites any existing value.  On error returns
    /// [`FeiMingError::Storage`] with a human-readable message.
    async fn put_seed(&self, key: &str, seed: &JiZhong) -> Result<()>;

    /// Retrieve the [`JiZhong`] stored under `key`, if any.
    ///
    /// Returns `Ok(None)` when the key does not exist — this is **not** an
    /// error condition.
    async fn get_seed(&self, key: &str) -> Result<Option<JiZhong>>;

    /// Delete the seed associated with `key`.
    ///
    /// Silently succeeds when the key does not exist.
    async fn remove_seed(&self, key: &str) -> Result<()>;
}

// ---------------------------------------------------------------------------
// DiCang
// ---------------------------------------------------------------------------

/// RocksDB-backed seed store with zstd compression and bincode serialization.
///
/// Each [`JiZhong`] is serialized with `bincode`, compressed with zstd at
/// [`ZSTD_YASUO_DENGJI`], and stored as a single RocksDB value.  The sync
/// RocksDB I/O is offloaded via `tokio::task::block_in_place` so the async
/// runtime is not blocked.
pub struct DiCang {
    db: DBWithThreadMode<MultiThreaded>,
}

impl DiCang {
    /// Open a RocksDB database at the given path.
    ///
    /// Configures the database with:
    /// * `create_if_missing = true`
    /// * parallelism scaled to `num_cpus`
    /// * up to 4 background compaction / flush threads
    /// * Snappy block compression at the SST level (independent of the
    ///   application-level zstd compression we apply on values)
    /// * missing column families created on open
    ///
    /// # Errors
    ///
    /// Returns [`FeiMingError::Storage`] if the database cannot be opened.
    pub fn new(path: &str) -> Result<Self> {
        let mut opts = RocksOptions::default();
        opts.create_if_missing(true);
        opts.increase_parallelism(num_cpus::get() as i32);
        opts.set_max_background_jobs(4);
        opts.set_compression_type(rocksdb::DBCompressionType::Snappy);
        opts.create_missing_column_families(true);
        let db = DBWithThreadMode::<MultiThreaded>::open(&opts, path)
            .map_err(|e| FeiMingError::Storage(format!("failed to open RocksDB at {}: {}", path, e)))?;
        info!(path = %path, "DiCang opened");
        Ok(Self { db })
    }

    /// Access the underlying database (for advanced operations).
    pub fn inner_db(&self) -> &DBWithThreadMode<MultiThreaded> {
        &self.db
    }
}

#[async_trait]
impl JiZhongCang for DiCang {
    #[instrument(level = "debug", skip(self, seed))]
    async fn put_seed(&self, key: &str, seed: &JiZhong) -> Result<()> {
        let key_buf = key.as_bytes().to_vec();

        block_in_place(|| {
            let encoded = bincode_encode(seed).map_err(|e| {
                FeiMingError::Storage(format!(
                    "bincode serialize failed for key '{}': {}",
                    key, e
                ))
            })?;

            let compressed = zstd_compress(&encoded).map_err(|e| {
                FeiMingError::Storage(format!(
                    "zstd compress failed for key '{}': {}",
                    key, e
                ))
            })?;

            self.db
                .put(&key_buf, &compressed)
                .map_err(|e| {
                    FeiMingError::Storage(format!(
                        "RocksDB put failed for key '{}': {}",
                        key, e
                    ))
                })?;

            debug!(key = %key, bytes = compressed.len(), "seed stored in RocksDB");
            Ok(())
        })
    }

    #[instrument(level = "debug", skip(self))]
    async fn get_seed(&self, key: &str) -> Result<Option<JiZhong>> {
        let key_buf = key.as_bytes().to_vec();

        block_in_place(|| {
            match self.db.get(&key_buf) {
                Ok(Some(raw)) => {
                    let decompressed = zstd_decompress(&raw).map_err(|e| {
                        FeiMingError::Storage(format!(
                            "zstd decompress failed for key '{}': {}",
                            key, e
                        ))
                    })?;

                    let seed: JiZhong = bincode_decode(&decompressed).map_err(|e| {
                        FeiMingError::Storage(format!(
                            "bincode deserialize failed for key '{}': {}",
                            key, e
                        ))
                    })?;

                    debug!(key = %key, "seed retrieved from RocksDB");
                    Ok(Some(seed))
                }
                Ok(None) => {
                    debug!(key = %key, "seed not found in RocksDB");
                    Ok(None)
                }
                Err(e) => Err(FeiMingError::Storage(format!(
                    "RocksDB get failed for key '{}': {}",
                    key, e
                ))),
            }
        })
    }

    #[instrument(level = "debug", skip(self))]
    async fn remove_seed(&self, key: &str) -> Result<()> {
        let key_buf = key.as_bytes().to_vec();

        block_in_place(|| {
            self.db
                .delete(&key_buf)
                .map_err(|e| {
                    FeiMingError::Storage(format!(
                        "RocksDB delete failed for key '{}': {}",
                        key, e
                    ))
                })?;

            debug!(key = %key, "seed removed from RocksDB");
            Ok(())
        })
    }
}

// ---------------------------------------------------------------------------
// TianCang
// ---------------------------------------------------------------------------

/// S3-backed seed store with zstd compression and bincode serialization.
///
/// Each [`JiZhong`] is serialized with `bincode`, compressed with zstd at
/// [`ZSTD_YASUO_DENGJI`], and uploaded as an S3 object.  An optional `prefix`
/// is prepended to every logical key so multiple environments or tenants can
/// share a single bucket safely.
pub struct TianCang {
    client: S3Client,
    bucket: String,
    prefix: Option<String>,
}

impl TianCang {
    /// Create a new `TianCang`.
    ///
    /// # Arguments
    ///
    /// * `client` — a pre-configured `aws_sdk_s3::Client`.
    /// * `bucket` — target S3 bucket name.
    /// * `prefix` — optional key prefix (e.g. `"jianai/mozhi/v1"`).  A trailing
    ///   slash is stripped automatically.
    pub fn new(client: S3Client, bucket: String, prefix: Option<String>) -> Self {
        Self { client, bucket, prefix }
    }

    /// Compute the full S3 object key from a logical key.
    ///
    /// If `prefix` is set, it is concatenated with `/` before the logical key.
    fn key(&self, key: &str) -> String {
        match &self.prefix {
            Some(p) => format!("{}/{}", p.trim_end_matches('/'), key),
            None => key.to_string(),
        }
    }
}

#[async_trait]
impl JiZhongCang for TianCang {
    #[instrument(level = "debug", skip(self, seed))]
    async fn put_seed(&self, key: &str, seed: &JiZhong) -> Result<()> {
        let object_key = self.key(key);

        let encoded = bincode_encode(seed).map_err(|e| {
            FeiMingError::Storage(format!(
                "bincode serialize failed for S3 key '{}': {}",
                object_key, e
            ))
        })?;

        let compressed = zstd_compress(&encoded).map_err(|e| {
            FeiMingError::Storage(format!(
                "zstd compress failed for S3 key '{}': {}",
                object_key, e
            ))
        })?;

        let body = ByteStream::from(compressed);

        self.client
            .put_object()
            .bucket(&self.bucket)
            .key(&object_key)
            .body(body)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Storage(format!(
                    "S3 put_object failed for bucket='{}' key='{}': {}",
                    self.bucket, object_key, e
                ))
            })?;

        debug!(bucket = %self.bucket, key = %object_key, "seed stored in S3");
        Ok(())
    }

    #[instrument(level = "debug", skip(self))]
    async fn get_seed(&self, key: &str) -> Result<Option<JiZhong>> {
        let object_key = self.key(key);

        let resp = self
            .client
            .get_object()
            .bucket(&self.bucket)
            .key(&object_key)
            .send()
            .await;

        match resp {
            Ok(output) => {
                let bytes = output
                    .body
                    .collect()
                    .await
                    .map_err(|e| {
                        FeiMingError::Storage(format!(
                            "S3 body stream collect failed for key '{}': {}",
                            object_key, e
                        ))
                    })?
                    .into_bytes();

                let decompressed = zstd_decompress(&bytes).map_err(|e| {
                    FeiMingError::Storage(format!(
                        "zstd decompress failed for S3 key '{}': {}",
                        object_key, e
                    ))
                })?;

                let seed: JiZhong = bincode_decode(&decompressed).map_err(|e| {
                    FeiMingError::Storage(format!(
                        "bincode deserialize failed for S3 key '{}': {}",
                        object_key, e
                    ))
                })?;

                debug!(bucket = %self.bucket, key = %object_key, "seed retrieved from S3");
                Ok(Some(seed))
            }
            Err(SdkError::ServiceError(ref service_err)) => {
                if service_err.err().is_no_such_key() {
                    debug!(bucket = %self.bucket, key = %object_key, "seed not found in S3 (NoSuchKey)");
                    Ok(None)
                } else {
                    Err(FeiMingError::Storage(format!(
                        "S3 get_object service error for bucket='{}' key='{}': {:?}",
                        self.bucket, object_key, service_err
                    )))
                }
            }
            Err(SdkError::DispatchFailure(ref dispatch_err)) => {
                warn!(bucket = %self.bucket, key = %object_key, error = ?dispatch_err, "S3 dispatch failure");
                Err(FeiMingError::Network(format!(
                    "S3 get_object dispatch failure for bucket='{}' key='{}': {:?}",
                    self.bucket, object_key, dispatch_err
                )))
            }
            Err(other) => {
                Err(FeiMingError::Storage(format!(
                    "S3 get_object failed for bucket='{}' key='{}': {}",
                    self.bucket, object_key, other
                )))
            }
        }
    }

    #[instrument(level = "debug", skip(self))]
    async fn remove_seed(&self, key: &str) -> Result<()> {
        let object_key = self.key(key);

        self.client
            .delete_object()
            .bucket(&self.bucket)
            .key(&object_key)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Storage(format!(
                    "S3 delete_object failed for bucket='{}' key='{}': {}",
                    self.bucket, object_key, e
                ))
            })?;

        debug!(bucket = %self.bucket, key = %object_key, "seed removed from S3");
        Ok(())
    }
}

// ---------------------------------------------------------------------------
// CangFa trait

/// Abstraction over a single cache storage tier.
///
/// Every tier (L1 memory, L2 RocksDB, L3 S3) implements this trait so the
/// [`JiGuanCheng`] can treat them uniformly.
#[async_trait]
pub trait CangFa: Send + Sync {
    /// Retrieve a [`MoYin`] by its string key.
    ///
    /// Returns `Ok(None)` on a cache miss.  TTL-aware backends should
    /// invalidate expired entries and report the miss.
    async fn get(&self, key: &str) -> Result<Option<MoYin>>;

    /// Persist a [`MoYin`] into this tier.
    async fn put(&self, entry: &MoYin) -> Result<()>;

    /// Remove a single entry identified by `key`.
    async fn delete(&self, key: &str) -> Result<()>;

    /// Remove **all** entries belonging to the given `model_id`.
    ///
    /// Returns the number of entries that were deleted.
    async fn delete_model(&self, model_id: &str) -> Result<u64>;

    /// Best-effort estimate of the total byte size of this tier.
    fn estimated_size_bytes(&self) -> u64;

    /// Best-effort estimate of the number of entries in this tier.
    fn estimated_entry_count(&self) -> u64;

    /// Allow down-casting to concrete types so that [`JiGuanCheng`] can
    /// aggregate per-tier statistics (hits / misses / evictions).
    fn as_any(&self) -> &dyn std::any::Any;
}

/// Outcome of a [`JiGuanCheng::lookup`] operation.
///
/// Encapsulates both the optional cached entry and the tier that served it,
/// enabling cache-hit telemetry and upstream backfill decisions.
#[derive(Debug, Clone)]
pub struct YinYing {
    /// The cached entry when a hit occurred.
    pub entry: Option<MoYin>,
    /// Which tier (or origin) produced this result.
    pub served_from: YinYuan,
}

impl YinYing {
    /// Construct a cache-miss result.
    pub fn miss() -> Self {
        Self {
            entry: None,
            served_from: YinYuan::Origin,
        }
    }

    /// Construct a cache-hit result.
    pub fn hit(entry: MoYin, served_from: YinYuan) -> Self {
        Self {
            entry: Some(entry),
            served_from,
        }
    }
}

// ---------------------------------------------------------------------------
// ZhuQue — Moka-based in-process cache
// ---------------------------------------------------------------------------

/// L1 cache backed by the [Moka](https://github.com/moka-rs/moka) high-performance
/// concurrent cache.
///
/// Provides sub-millisecond latency for hot inference results.  Entries are
/// evicted automatically by Moka when the capacity bound is exceeded or the
/// TTL expires.
pub struct ZhuQue {
    cache: MokaCache<String, MoYin>,
    hits: AtomicU64,
    misses: AtomicU64,
    evictions: AtomicU64,
}

impl ZhuQue {
    /// Create a new L1 in-memory cache.
    ///
    /// # Arguments
    /// * `capacity` — Maximum number of entries before eviction kicks in.
    /// * `ttl_secs` — Time-to-live in seconds.  Use `0` for no TTL.
    pub fn new(capacity: u64, ttl_secs: u64) -> Self {
        let hits = AtomicU64::new(0);
        let misses = AtomicU64::new(0);
        let evictions = AtomicU64::new(0);

        let evictions_clone = AtomicU64::new(0);

        let mut builder = MokaCache::builder().max_capacity(capacity);
        if ttl_secs > 0 {
            builder = builder.time_to_live(Duration::from_secs(ttl_secs));
        }

        let cache = builder
            .eviction_listener(move |_key, _value, _cause| {
                evictions_clone.fetch_add(1, Ordering::Relaxed);
            })
            .build();

        Self {
            cache,
            hits,
            misses,
            evictions,
        }
    }

    /// Peek at an entry without updating the internal hit counter.
    ///
    /// Useful for diagnostics or pre-flight checks.
    pub fn peek(&self, key: &str) -> Option<MoYin> {
        self.cache.get(key)
    }

    /// Return the raw `(hits, misses, evictions)` counters for this tier.
    pub fn stats(&self) -> (u64, u64, u64) {
        (
            self.hits.load(Ordering::Relaxed),
            self.misses.load(Ordering::Relaxed),
            self.evictions.load(Ordering::Relaxed),
        )
    }
}

#[async_trait]
impl CangFa for ZhuQue {
    async fn get(&self, key: &str) -> Result<Option<MoYin>> {
        match self.cache.get(key) {
            Some(entry) => {
                if entry.expires_at > 0 && entry.is_expired() {
                    self.cache.invalidate(key);
                    self.misses.fetch_add(1, Ordering::Relaxed);
                    Ok(None)
                } else {
                    self.hits.fetch_add(1, Ordering::Relaxed);
                    Ok(Some(entry))
                }
            }
            None => {
                self.misses.fetch_add(1, Ordering::Relaxed);
                Ok(None)
            }
        }
    }

    async fn put(&self, entry: &MoYin) -> Result<()> {
        self.cache.insert(entry.key.clone(), entry.clone());
        Ok(())
    }

    async fn delete(&self, key: &str) -> Result<()> {
        self.cache.invalidate(key);
        Ok(())
    }

    async fn delete_model(&self, model_id: &str) -> Result<u64> {
        let keys_to_remove: Vec<String> = self
            .cache
            .iter()
            .filter(|(_k, v)| v.model_id == model_id)
            .map(|(k, _v)| (*k).clone())
            .collect();

        let count = keys_to_remove.len() as u64;
        for k in keys_to_remove {
            self.cache.invalidate(&k);
        }

        if count > 0 {
            self.evictions.fetch_add(count, Ordering::Relaxed);
        }

        Ok(count)
    }

    fn estimated_size_bytes(&self) -> u64 {
        self.cache
            .iter()
            .map(|(k, v)| k.len() as u64 + v.output_data.len() as u64 + 64)
            .sum()
    }

    fn estimated_entry_count(&self) -> u64 {
        self.cache.entry_count()
    }

    fn as_any(&self) -> &dyn std::any::Any {
        self
    }
}

// ---------------------------------------------------------------------------
// XuanWu — local on-disk persistent cache
// ---------------------------------------------------------------------------

/// L2 cache backed by RocksDB.
///
/// RocksDB is configured with LZ4 compression, bounded file descriptors, and
/// tuned write-buffer settings for sustained write throughput under inference
/// load.  All DB operations are guarded by an [`RwLock`] to satisfy the
/// `CangFa` `Send + Sync` contract.
pub struct XuanWu {
    db: std::sync::RwLock<DBWithThreadMode<MultiThreaded>>,
    hits: AtomicU64,
    misses: AtomicU64,
    evictions: AtomicU64,
}

impl XuanWu {
    /// Open (or create) a RocksDB instance at `path` for use as the L2 cache.
    ///
    /// # Errors
    /// Returns [`FeiMingError::Storage`] if RocksDB cannot be opened.
    pub fn new(path: &str) -> Result<Self> {
        let mut opts = RocksOptions::default();
        opts.set_compression_type(rocksdb::DBCompressionType::None);
        opts.set_max_open_files(LUYIN_MAX_WENJIAN as i32);
        opts.set_write_buffer_size(LUYIN_XIERU_HUANCHONG as usize);
        opts.set_max_write_buffer_number(3);
        opts.set_min_write_buffer_number_to_merge(1);
        opts.create_if_missing(true);

        let db =
            DBWithThreadMode::<MultiThreaded>::open(&opts, path).map_err(|e| {
                FeiMingError::Storage(format!("RocksDB open failed at '{}': {}", path, e))
            })?;

        Ok(Self {
            db: std::sync::RwLock::new(db),
            hits: AtomicU64::new(0),
            misses: AtomicU64::new(0),
            evictions: AtomicU64::new(0),
        })
    }

    /// Return the raw `(hits, misses, evictions)` counters for this tier.
    pub fn stats(&self) -> (u64, u64, u64) {
        (
            self.hits.load(Ordering::Relaxed),
            self.misses.load(Ordering::Relaxed),
            self.evictions.load(Ordering::Relaxed),
        )
    }

    /// Convert a logical cache key into the byte key stored in RocksDB.
    fn make_db_key(key: &str) -> Vec<u8> {
        format!("cache:{}", key).into_bytes()
    }
}

#[async_trait]
impl CangFa for XuanWu {
    async fn get(&self, key: &str) -> Result<Option<MoYin>> {
        let db_key = Self::make_db_key(key);
        let db_guard = self.db.read().unwrap_or_else(|e| e.into_inner());

        match db_guard.get(&db_key) {
            Ok(Some(data)) => {
                drop(db_guard);
                let entry: MoYin = bincode_decode(&data).map_err(|e| {
                    FeiMingError::Serialization(format!(
                        "L2 bincode deserialize failed for key '{}': {}",
                        key, e
                    ))
                })?;

                if entry.expires_at > 0 && entry.is_expired() {
                    let write_guard = self.db.write().unwrap_or_else(|e| e.into_inner());
                    write_guard
                        .delete(&db_key)
                        .map_err(|e| FeiMingError::Storage(format!("RocksDB delete failed: {}", e)))?;
                    self.misses.fetch_add(1, Ordering::Relaxed);
                    self.evictions.fetch_add(1, Ordering::Relaxed);
                    Ok(None)
                } else {
                    self.hits.fetch_add(1, Ordering::Relaxed);
                    Ok(Some(entry))
                }
            }
            Ok(None) => {
                drop(db_guard);
                self.misses.fetch_add(1, Ordering::Relaxed);
                Ok(None)
            }
            Err(e) => {
                drop(db_guard);
                Err(FeiMingError::Storage(format!(
                    "RocksDB get failed for key '{}': {}",
                    key, e
                )))
            }
        }
    }

    async fn put(&self, entry: &MoYin) -> Result<()> {
        let db_key = Self::make_db_key(&entry.key);
        let data = bincode_encode(entry).map_err(|e| {
            FeiMingError::Serialization(format!(
                "L2 bincode serialize failed for key '{}': {}",
                entry.key, e
            ))
        })?;
        let db_guard = self.db.read().unwrap_or_else(|e| e.into_inner());
        db_guard
            .put(&db_key, data)
            .map_err(|e| FeiMingError::Storage(format!("RocksDB put failed: {}", e)))?;
        Ok(())
    }

    async fn delete(&self, key: &str) -> Result<()> {
        let db_key = Self::make_db_key(key);
        let db_guard = self.db.read().unwrap_or_else(|e| e.into_inner());
        db_guard
            .delete(&db_key)
            .map_err(|e| FeiMingError::Storage(format!("RocksDB delete failed: {}", e)))?;
        Ok(())
    }

    async fn delete_model(&self, model_id: &str) -> Result<u64> {
        let prefix = b"cache:";
        let db_guard = self.db.read().unwrap_or_else(|e| e.into_inner());
        let mut keys_to_delete: Vec<Vec<u8>> = Vec::new();

        let mut iter = db_guard.raw_iterator();
        iter.seek_to_first();
        while iter.valid() {
            if let (Some(key), Some(value)) = (iter.key(), iter.value()) {
                if key.starts_with(prefix) {
                    if let Ok(entry) = bincode_decode::<MoYin>(value) {
                        if entry.model_id == model_id {
                            keys_to_delete.push(key.to_vec());
                        }
                    }
                }
            }
            iter.next();
        }
        // Explicitly drop the iterator and read guard before acquiring write lock.
        drop(iter);
        drop(db_guard);

        let count = keys_to_delete.len() as u64;
        if count > 0 {
            let write_guard = self.db.write().unwrap_or_else(|e| e.into_inner());
            for (i, key) in keys_to_delete.iter().enumerate() {
                write_guard
                    .delete(key)
                    .map_err(|e| FeiMingError::Storage(format!("RocksDB batch delete failed: {}", e)))?;
                if (i + 1) % LUYIN_PIHANGSHUAXIN_CI as usize == 0 {
                    write_guard
                        .flush()
                        .map_err(|e| FeiMingError::Storage(format!("RocksDB flush failed: {}", e)))?;
                }
            }
            write_guard
                .flush()
                .map_err(|e| FeiMingError::Storage(format!("RocksDB final flush failed: {}", e)))?;
            self.evictions.fetch_add(count, Ordering::Relaxed);
        }

        Ok(count)
    }

    fn estimated_size_bytes(&self) -> u64 {
        let db_guard = self.db.read().unwrap_or_else(|e| e.into_inner());
        db_guard
            .property_int_value("rocksdb.estimate-live-data-size")
            .unwrap_or(None)
            .unwrap_or(0)
    }

    fn estimated_entry_count(&self) -> u64 {
        let db_guard = self.db.read().unwrap_or_else(|e| e.into_inner());
        db_guard
            .property_int_value("rocksdb.estimate-num-keys")
            .unwrap_or(None)
            .unwrap_or(0)
    }

    fn as_any(&self) -> &dyn std::any::Any {
        self
    }
}

// ---------------------------------------------------------------------------
// QingLong — remote S3-backed cache
// ---------------------------------------------------------------------------

/// L3 cache backed by Amazon S3 (or S3-compatible object stores).
///
/// Entries are stored as individually compressed objects under a configurable
/// prefix.  A [`Semaphore`] caps concurrent S3 operations to prevent
/// thundering-herd against the object-store API under cache-miss storms.
pub struct QingLong {
    client: S3Client,
    bucket: String,
    prefix: String,
    semaphore: Arc<Semaphore>,
    hits: AtomicU64,
    misses: AtomicU64,
    evictions: AtomicU64,
}

impl QingLong {
    /// Create a new L3 S3 cache instance.
    ///
    /// # Arguments
    /// * `client` — Pre-configured AWS S3 client.
    /// * `bucket` — Target S3 bucket name.
    /// * `prefix` — Optional key prefix.  Empty string means root.
    pub fn new(client: S3Client, bucket: String, prefix: Option<String>) -> Self {
        let prefix = prefix
            .map(|p| {
                let trimmed = p.trim().trim_matches('/');
                if trimmed.is_empty() {
                    String::new()
                } else {
                    format!("{}/", trimmed)
                }
            })
            .unwrap_or_default();

        Self {
            client,
            bucket,
            prefix,
            semaphore: Arc::new(Semaphore::new(YUNYIN_BINGFA_SHANG)),
            hits: AtomicU64::new(0),
            misses: AtomicU64::new(0),
            evictions: AtomicU64::new(0),
        }
    }

    /// Compute the S3 object key for a logical cache key.
    fn s3_key(&self, key: &str) -> String {
        let safe = key
            .trim()
            .replace("//", "/")
            .replace('\\', "/");
        format!("{}cache/{}", self.prefix, safe)
    }

    /// Compute the S3 prefix used when listing objects for a specific model.
    fn s3_prefix_for_model(&self, model_id: &str) -> String {
        let sanitized = model_id
            .trim()
            .replace("//", "/")
            .replace('\\', "/");
        format!("{}cache/model:{}/", self.prefix, sanitized)
    }

    /// Return the raw `(hits, misses, evictions)` counters for this tier.
    pub fn stats(&self) -> (u64, u64, u64) {
        (
            self.hits.load(Ordering::Relaxed),
            self.misses.load(Ordering::Relaxed),
            self.evictions.load(Ordering::Relaxed),
        )
    }
}

#[async_trait]
impl CangFa for QingLong {
    async fn get(&self, key: &str) -> Result<Option<MoYin>> {
        let _permit = self
            .semaphore
            .acquire()
            .await
            .map_err(|e| FeiMingError::Network(format!("S3 semaphore acquire failed: {}", e)))?;

        let s3_key = self.s3_key(key);

        match self
            .client
            .get_object()
            .bucket(&self.bucket)
            .key(&s3_key)
            .send()
            .await
        {
            Ok(output) => {
                let bytes = output
                    .body
                    .collect()
                    .await
                    .map_err(|e| FeiMingError::Network(format!("S3 body collect failed: {}", e)))?
                    .into_bytes();

                let decompressed = zstd_decompress(&bytes).map_err(|e| {
                    FeiMingError::Serialization(format!(
                        "S3 zstd decompress failed for key '{}': {}",
                        key, e
                    ))
                })?;

                let entry: MoYin = bincode_decode(&decompressed).map_err(|e| {
                    FeiMingError::Serialization(format!(
                        "S3 bincode deserialize failed for key '{}': {}",
                        key, e
                    ))
                })?;

                if entry.expires_at > 0 && entry.is_expired() {
                    drop(_permit);
                    if let Err(e) = self.delete(key).await {
                        warn!(
                            "Failed to delete expired S3 cache entry '{}': {}",
                            key, e
                        );
                    }
                    self.misses.fetch_add(1, Ordering::Relaxed);
                    self.evictions.fetch_add(1, Ordering::Relaxed);
                    Ok(None)
                } else {
                    self.hits.fetch_add(1, Ordering::Relaxed);
                    Ok(Some(entry))
                }
            }
            Err(SdkError::ServiceError(ref se)) => {
                let code = se.err().meta().code().unwrap_or("");
                if code == "NoSuchKey" || code == "404" {
                    self.misses.fetch_add(1, Ordering::Relaxed);
                    Ok(None)
                } else {
                    Err(FeiMingError::Network(format!(
                        "S3 get service error for key '{}': {:?}",
                        key, se
                    )))
                }
            }
            Err(other) => Err(FeiMingError::Network(format!(
                "S3 get error for key '{}': {}",
                key, other
            ))),
        }
    }

    async fn put(&self, entry: &MoYin) -> Result<()> {
        let _permit = self
            .semaphore
            .acquire()
            .await
            .map_err(|e| FeiMingError::Network(format!("S3 semaphore acquire failed: {}", e)))?;

        let s3_key = self.s3_key(&entry.key);
        let data = bincode_encode(entry).map_err(|e| {
            FeiMingError::Serialization(format!(
                "S3 bincode serialize failed for key '{}': {}",
                entry.key, e
            ))
        })?;

        let compressed = zstd_compress(&data).map_err(|e| {
            FeiMingError::Serialization(format!(
                "S3 zstd compress failed for key '{}': {}",
                entry.key, e
            ))
        })?;

        let body = ByteStream::from(compressed);

        self.client
            .put_object()
            .bucket(&self.bucket)
            .key(&s3_key)
            .body(body)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!(
                    "S3 put_object failed for key '{}': {}",
                    s3_key, e
                ))
            })?;

        Ok(())
    }

    async fn delete(&self, key: &str) -> Result<()> {
        let _permit = self
            .semaphore
            .acquire()
            .await
            .map_err(|e| FeiMingError::Network(format!("S3 semaphore acquire failed: {}", e)))?;

        let s3_key = self.s3_key(key);
        self.client
            .delete_object()
            .bucket(&self.bucket)
            .key(&s3_key)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!(
                    "S3 delete_object failed for key '{}': {}",
                    s3_key, e
                ))
            })?;

        Ok(())
    }

    async fn delete_model(&self, model_id: &str) -> Result<u64> {
        let _permit = self
            .semaphore
            .acquire()
            .await
            .map_err(|e| FeiMingError::Network(format!("S3 semaphore acquire failed: {}", e)))?;

        let prefix = self.s3_prefix_for_model(model_id);
        let mut keys_to_delete: Vec<String> = Vec::new();
        let mut continuation_token: Option<String> = None;

        loop {
            let mut request = self
                .client
                .list_objects_v2()
                .bucket(&self.bucket)
                .prefix(&prefix);
            if let Some(token) = continuation_token {
                request = request.continuation_token(token);
            }

            let resp = request.send().await.map_err(|e| {
                FeiMingError::Network(format!(
                    "S3 list_objects_v2 failed for model '{}': {}",
                    model_id, e
                ))
            })?;

            if let Some(contents) = resp.contents {
                for obj in contents {
                    if let Some(key) = obj.key {
                        keys_to_delete.push(key);
                    }
                }
            }

            if resp.is_truncated.unwrap_or(false) {
                continuation_token = resp.next_continuation_token;
            } else {
                break;
            }
        }

        let count = keys_to_delete.len() as u64;
        for key in &keys_to_delete {
            self.client
                .delete_object()
                .bucket(&self.bucket)
                .key(key)
                .send()
                .await
                .map_err(|e| {
                    FeiMingError::Network(format!(
                        "S3 delete_object failed for key '{}': {}",
                        key, e
                    ))
                })?;
        }

        if count > 0 {
            self.evictions.fetch_add(count, Ordering::Relaxed);
        }

        Ok(count)
    }

    fn estimated_size_bytes(&self) -> u64 {
        // S3 does not expose a cheap per-prefix size API.
        0
    }

    fn estimated_entry_count(&self) -> u64 {
        // Counting keys via ListObjectsV2 is expensive; return 0 as a placeholder.
        0
    }

    fn as_any(&self) -> &dyn std::any::Any {
        self
    }
}

/// Top-level cache facade that coordinates lookups and writes across all tiers.
///
/// # Lookup semantics
/// 1. **L1** — Fastest path.  If hit, return immediately.
/// 2. **L2** — Local disk.  If hit, *backfill* entry into L1, then return.
/// 3. **L3** — Remote S3.  If hit, *backfill* entry into L1 **and** L2, then return.
/// 4. **Miss** — Return [`YinYing::miss()`]; caller should run inference.
///
/// # Insert semantics
/// - L1 and L2 are written **synchronously** so subsequent lookups see the
///   result immediately.
/// - L3 is written **asynchronously** (fire-and-forget via `tokio::spawn`) to
///   avoid adding S3 latency to the hot path.
///
/// # Eviction semantics
/// - `remove` / `delete` propagates to **all** tiers.
/// - `evict_model` scans and deletes every entry belonging to the given model
///   from all tiers, returning the total count of removed entries.
pub struct JiGuanCheng {
    l1: Arc<dyn CangFa>,
    l2: Arc<dyn CangFa>,
    l3: Option<Arc<dyn CangFa>>,
}

impl JiGuanCheng {
    /// Construct a [`JiGuanCheng`] from the three tiers.
    ///
    /// `l3` is optional — when `None`, the system operates as a two-tier
    /// memory + disk cache.
    pub fn new(
        l1: Arc<dyn CangFa>,
        l2: Arc<dyn CangFa>,
        l3: Option<Arc<dyn CangFa>>,
    ) -> Self {
        Self { l1, l2, l3 }
    }

    /// Multi-tier lookup with automatic backfill.
    ///
    /// Searches L1 → L2 → L3 in order.  On an L2 hit the entry is cloned into
    /// L1.  On an L3 hit the entry is cloned into both L1 and L2.  These
    /// backfill operations are best-effort — failures are logged as warnings
    /// but do not fail the overall lookup.
    pub async fn lookup(&self, key: &str) -> Result<YinYing> {
        // --- L1 ---
        match self.l1.get(key).await? {
            Some(entry) => return Ok(YinYing::hit(entry, YinYuan::L1Memory)),
            None => {}
        }

        // --- L2 ---
        match self.l2.get(key).await? {
            Some(entry) => {
                if let Err(e) = self.l1.put(&entry).await {
                    warn!("Failed to backfill L1 from L2 for key '{}': {}", key, e);
                }
                return Ok(YinYing::hit(entry, YinYuan::L2RocksDb));
            }
            None => {}
        }

        // --- L3 ---
        if let Some(ref l3) = self.l3 {
            match l3.get(key).await? {
                Some(entry) => {
                    if let Err(e) = self.l1.put(&entry).await {
                        warn!(
                            "Failed to backfill L1 from L3 for key '{}': {}",
                            key, e
                        );
                    }
                    if let Err(e) = self.l2.put(&entry).await {
                        warn!(
                            "Failed to backfill L2 from L3 for key '{}': {}",
                            key, e
                        );
                    }
                    return Ok(YinYing::hit(entry, YinYuan::L3S3));
                }
                None => {}
            }
        }

        // --- Miss ---
        Ok(YinYing::miss())
    }

    /// Insert an entry into the cache tiers.
    ///
    /// L1 and L2 receive the entry synchronously.  L3 (if configured) receives
    /// it asynchronously via a spawned task so that S3 latency never blocks
    /// the inference hot path.
    pub async fn insert(&self, entry: &MoYin) -> Result<()> {
        self.l1.put(entry).await?;
        self.l2.put(entry).await?;

        if let Some(ref l3) = self.l3 {
            let l3_clone = Arc::clone(l3);
            let entry_clone = entry.clone();
            tokio::spawn(async move {
                if let Err(e) = l3_clone.put(&entry_clone).await {
                    warn!(
                        "Async L3 cache write failed for key '{}': {}",
                        entry_clone.key, e
                    );
                }
            });
        }

        Ok(())
    }

    /// Remove a single entry from **all** tiers.
    ///
    /// Errors from individual tiers are logged but not propagated upward, so
    /// the operation always reports success at the top level.
    pub async fn remove(&self, key: &str) -> Result<()> {
        if let Err(e) = self.l1.delete(key).await {
            warn!("L1 delete failed for key '{}': {}", key, e);
        }
        if let Err(e) = self.l2.delete(key).await {
            warn!("L2 delete failed for key '{}': {}", key, e);
        }
        if let Some(ref l3) = self.l3 {
            if let Err(e) = l3.delete(key).await {
                warn!("L3 delete failed for key '{}': {}", key, e);
            }
        }
        Ok(())
    }

    /// Evict every entry belonging to `model_id` from all tiers.
    ///
    /// Returns the total number of entries removed across L1, L2, and L3.
    /// If any tier fails the error is propagated after the others have been
    /// attempted.
    pub async fn evict_model(&self, model_id: &str) -> Result<u64> {
        let l1_count = self.l1.delete_model(model_id).await.map_err(|e| {
            FeiMingError::Cache(format!(
                "L1 evict_model failed for model '{}': {}",
                model_id, e
            ))
        })?;
        let l2_count = self.l2.delete_model(model_id).await.map_err(|e| {
            FeiMingError::Cache(format!(
                "L2 evict_model failed for model '{}': {}",
                model_id, e
            ))
        })?;
        let l3_count = match &self.l3 {
            Some(l3) => l3.delete_model(model_id).await.map_err(|e| {
                FeiMingError::Network(format!(
                    "L3 evict_model failed for model '{}': {}",
                    model_id, e
                ))
            })?,
            None => 0,
        };
        Ok(l1_count + l2_count + l3_count)
    }

    /// Aggregate statistics from all tiers.
    ///
    /// Uses down-casting via [`CangFa::as_any`] to read the concrete
    /// tier counters.  `total_hits` and `total_misses` are computed sums.
    pub fn stats(&self) -> ChengZhi {
        let mut s = ChengZhi::default();

        // L1 counters & estimates
        if let Some(l1) = self.l1.as_any().downcast_ref::<ZhuQue>() {
            let (h, m, e) = l1.stats();
            s.l1_hits = h;
            s.l1_misses = m;
            s.l1_evictions = e;
        }
        s.l1_size_bytes = self.l1.estimated_size_bytes();
        s.l1_entry_count = self.l1.estimated_entry_count();

        // L2 counters & estimates
        if let Some(l2) = self.l2.as_any().downcast_ref::<XuanWu>() {
            let (h, m, e) = l2.stats();
            s.l2_hits = h;
            s.l2_misses = m;
            s.l2_evictions = e;
        }
        s.l2_size_bytes = self.l2.estimated_size_bytes();
        s.l2_entry_count = self.l2.estimated_entry_count();

        // L3 counters & estimates
        if let Some(ref l3) = self.l3 {
            if let Some(l3_inner) = l3.as_any().downcast_ref::<QingLong>() {
                let (h, m, e) = l3_inner.stats();
                s.l3_hits = h;
                s.l3_misses = m;
                s.l3_evictions = e;
            }
            s.l3_size_bytes = l3.estimated_size_bytes();
            s.l3_entry_count = l3.estimated_entry_count();
        }

        s.total_hits = s.l1_hits + s.l2_hits + s.l3_hits;
        s.total_misses = s.l1_misses + s.l2_misses + s.l3_misses;

        s
    }

    /// Inherent `get` shorthand — delegates to the [`CangFa`] trait impl.
    pub async fn get(&self, key: &str) -> Result<Option<MoYin>> {
        CangFa::get(self, key).await
    }

    /// Inherent `put` shorthand — delegates to the [`CangFa`] trait impl.
    pub async fn put(&self, entry: &MoYin) -> Result<()> {
        CangFa::put(self, entry).await
    }
}


#[async_trait]
impl CangFa for JiGuanCheng {
    /// Delegates to [`JiGuanCheng::lookup`] and extracts only the entry.
    async fn get(&self, key: &str) -> Result<Option<MoYin>> {
        let result = self.lookup(key).await?;
        Ok(result.entry)
    }

    /// Delegates to [`JiGuanCheng::insert`].
    async fn put(&self, entry: &MoYin) -> Result<()> {
        self.insert(entry).await
    }

    /// Delegates to [`JiGuanCheng::remove`].
    async fn delete(&self, key: &str) -> Result<()> {
        self.remove(key).await
    }

    /// Delegates to [`JiGuanCheng::evict_model`].
    async fn delete_model(&self, model_id: &str) -> Result<u64> {
        self.evict_model(model_id).await
    }

    /// Sum of `estimated_size_bytes` across L1, L2, and L3.
    fn estimated_size_bytes(&self) -> u64 {
        let mut total = self.l1.estimated_size_bytes();
        total += self.l2.estimated_size_bytes();
        if let Some(ref l3) = self.l3 {
            total += l3.estimated_size_bytes();
        }
        total
    }

    /// Sum of `estimated_entry_count` across L1, L2, and L3.
    fn estimated_entry_count(&self) -> u64 {
        let mut total = self.l1.estimated_entry_count();
        total += self.l2.estimated_entry_count();
        if let Some(ref l3) = self.l3 {
            total += l3.estimated_entry_count();
        }
        total
    }

    fn as_any(&self) -> &dyn std::any::Any {
        self
    }
}

/// Circuit breaker states: Closed (allowing), HalfOpen (testing), Open (rejecting).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum FeiGongTai {
    /// Normal operation — requests allowed.
    Closed = 0,
    /// Probing state — limited requests allowed to test recovery.
    HalfOpen = 1,
    /// Rejecting all requests after threshold failures.
    Open = 2,
}

impl FeiGongTai {
    /// Convert a raw `u32` into the corresponding `FeiGongTai`.
    /// Values outside 0..=2 map to `Closed` as a safe default.
    pub const fn from_u32(v: u32) -> Self {
        match v {
            0 => FeiGongTai::Closed,
            1 => FeiGongTai::HalfOpen,
            2 => FeiGongTai::Open,
            _ => FeiGongTai::Closed,
        }
    }
}

impl From<u32> for FeiGongTai {
    fn from(v: u32) -> Self {
        FeiGongTai::from_u32(v)
    }
}

impl fmt::Display for FeiGongTai {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            FeiGongTai::Closed => write!(f, "Closed"),
            FeiGongTai::HalfOpen => write!(f, "HalfOpen"),
            FeiGongTai::Open => write!(f, "Open"),
        }
    }
}

/// Production-grade circuit breaker using only atomic operations.
///
/// The circuit breaker starts in the `Closed` state, allowing all requests.
/// After `failure_threshold` consecutive failures, it transitions to `Open`,
/// rejecting all requests for `open_duration_secs`. After that timeout, it
/// transitions to `HalfOpen`, allowing a limited number of requests to test
/// recovery. After `half_open_successes` consecutive successes in `HalfOpen`,
/// it transitions back to `Closed`.
///
/// All operations are lock-free using `AtomicU32` / `AtomicU64`, making it
/// suitable for high-contention multi-threaded environments.
pub struct FeiGong {
    state: AtomicU32,
    failure_count: AtomicU64,
    consecutive_successes: AtomicU64,
    failure_threshold: u64,
    half_open_successes: u64,
    open_duration_secs: u64,
    last_opened_at: AtomicU64,
}

impl FeiGong {
    /// Create a new circuit breaker with the given thresholds.
    ///
    /// # Arguments
    /// * `failure_threshold` — number of consecutive failures before opening.
    /// * `open_duration_secs` — seconds to stay open before transitioning to half-open.
    /// * `half_open_successes` — consecutive successes in half-open to close.
    pub fn new(failure_threshold: u64, open_duration_secs: u64, half_open_successes: u64) -> Self {
        Self {
            state: AtomicU32::new(FeiGongTai::Closed as u32),
            failure_count: AtomicU64::new(0),
            consecutive_successes: AtomicU64::new(0),
            failure_threshold,
            half_open_successes,
            open_duration_secs,
            last_opened_at: AtomicU64::new(0),
        }
    }

    /// Check whether the circuit breaker permits a request.
    ///
    /// Returns `Ok(())` if the request may proceed, or
    /// `Err(FeiMingError::CircuitBreakerOpen)` if the circuit is open and the
    /// cooldown period has not yet elapsed.
    pub fn allow(&self) -> Result<()> {
        let state_u32 = self.state.load(Ordering::Acquire);
        let state = FeiGongTai::from_u32(state_u32);

        match state {
            FeiGongTai::Closed => Ok(()),
            FeiGongTai::HalfOpen => Ok(()),
            FeiGongTai::Open => {
                let last_opened = self.last_opened_at.load(Ordering::Acquire);
                let now = SystemTime::now()
                    .duration_since(UNIX_EPOCH)
                    .unwrap_or_default()
                    .as_secs();
                if now.saturating_sub(last_opened) < self.open_duration_secs {
                    return Err(FeiMingError::CircuitBreakerOpen);
                }

                // Attempt transition: Open -> HalfOpen
                match self.state.compare_exchange(
                    FeiGongTai::Open as u32,
                    FeiGongTai::HalfOpen as u32,
                    Ordering::AcqRel,
                    Ordering::Acquire,
                ) {
                    Ok(_) => {
                        self.consecutive_successes.store(0, Ordering::Release);
                        info!(
                            "circuit breaker transitioned Open -> HalfOpen (cooldown elapsed: {}s)",
                            self.open_duration_secs
                        );
                        Ok(())
                    }
                    Err(actual) => {
                        // Another thread won the CAS — re-evaluate the actual state.
                        let actual_state = FeiGongTai::from_u32(actual);
                        if actual_state == FeiGongTai::HalfOpen {
                            Ok(())
                        } else {
                            Err(FeiMingError::CircuitBreakerOpen)
                        }
                    }
                }
            }
        }
    }

    /// Record a successful operation.
    ///
    /// In `Closed` state, resets the failure counter.
    /// In `HalfOpen` state, increments the consecutive-success counter and
    /// transitions to `Closed` if enough successes have been recorded.
    /// In `Open` state, this is a no-op.
    pub fn record_success(&self) {
        let state_u32 = self.state.load(Ordering::Acquire);
        let state = FeiGongTai::from_u32(state_u32);

        match state {
            FeiGongTai::Closed => {
                // Reset failure count only if it is non-zero to avoid unnecessary writes.
                let prev = self.failure_count.swap(0, Ordering::Release);
                if prev > 0 {
                    debug!("circuit breaker reset failure_count (was {})", prev);
                }
            }
            FeiGongTai::HalfOpen => {
                let successes = self.consecutive_successes.fetch_add(1, Ordering::AcqRel) + 1;
                if successes >= self.half_open_successes {
                    // Best-effort transition HalfOpen -> Closed.
                    let _ = self.state.compare_exchange(
                        FeiGongTai::HalfOpen as u32,
                        FeiGongTai::Closed as u32,
                        Ordering::AcqRel,
                        Ordering::Acquire,
                    );
                    self.consecutive_successes.store(0, Ordering::Release);
                    self.failure_count.store(0, Ordering::Release);
                    info!(
                        "circuit breaker transitioned HalfOpen -> Closed ({} consecutive successes)",
                        successes
                    );
                }
            }
            FeiGongTai::Open => {
                // No-op: open circuit does not record successes.
            }
        }
    }

    /// Record a failed operation.
    ///
    /// In `Closed` state, increments the failure counter and opens the circuit
    /// if the threshold is reached.
    /// In `HalfOpen` state, immediately transitions back to `Open`.
    /// In `Open` state, this is a no-op.
    pub fn record_failure(&self) {
        let state_u32 = self.state.load(Ordering::Acquire);
        let state = FeiGongTai::from_u32(state_u32);

        match state {
            FeiGongTai::Closed => {
                let failures = self.failure_count.fetch_add(1, Ordering::AcqRel) + 1;
                if failures >= self.failure_threshold {
                    let now = SystemTime::now()
                        .duration_since(UNIX_EPOCH)
                        .unwrap_or_default()
                        .as_secs();
                    match self.state.compare_exchange(
                        FeiGongTai::Closed as u32,
                        FeiGongTai::Open as u32,
                        Ordering::AcqRel,
                        Ordering::Acquire,
                    ) {
                        Ok(_) => {
                            self.last_opened_at.store(now, Ordering::Release);
                            self.consecutive_successes.store(0, Ordering::Release);
                            warn!(
                                "circuit breaker transitioned Closed -> Open ({} failures, threshold {})",
                                failures, self.failure_threshold
                            );
                        }
                        Err(_) => {
                            // Another thread already transitioned; ignore.
                        }
                    }
                }
            }
            FeiGongTai::HalfOpen => {
                let now = SystemTime::now()
                    .duration_since(UNIX_EPOCH)
                    .unwrap_or_default()
                    .as_secs();
                match self.state.compare_exchange(
                    FeiGongTai::HalfOpen as u32,
                    FeiGongTai::Open as u32,
                    Ordering::AcqRel,
                    Ordering::Acquire,
                ) {
                    Ok(_) => {
                        self.last_opened_at.store(now, Ordering::Release);
                        self.consecutive_successes.store(0, Ordering::Release);
                        warn!("circuit breaker transitioned HalfOpen -> Open (failure in half-open)");
                    }
                    Err(_) => {
                        // Another thread already transitioned; ignore.
                    }
                }
            }
            FeiGongTai::Open => {
                // No-op: already open.
            }
        }
    }

    /// Return the current circuit breaker state.
    pub fn state(&self) -> FeiGongTai {
        FeiGongTai::from_u32(self.state.load(Ordering::Acquire))
    }

    /// Return the raw `u32` state value.
    pub fn state_u32(&self) -> u32 {
        self.state.load(Ordering::Acquire)
    }

    /// Return the current failure count.
    pub fn failure_count(&self) -> u64 {
        self.failure_count.load(Ordering::Acquire)
    }

    /// Return the current consecutive-success count (used in half-open probing).
    pub fn consecutive_successes(&self) -> u64 {
        self.consecutive_successes.load(Ordering::Acquire)
    }

    /// Return the Unix timestamp (seconds) when the circuit last opened, or 0.
    pub fn last_opened_at(&self) -> u64 {
        self.last_opened_at.load(Ordering::Acquire)
    }
}

/// Outcome of a deduplication attempt: either this caller becomes the leader
/// responsible for executing the operation, or it becomes a follower that
/// waits for the leader's result.
pub enum TongJu {
    /// This caller is the leader and must execute the operation, then broadcast.
    Leader(TongZhu),
    /// This caller is a follower and should wait on the receiver for the result.
    Follower(broadcast::Receiver<Result<Vec<u8>>>),
}

/// Leader handle for a deduplicated operation.
///
/// The leader executes the actual work and calls [`complete`] to broadcast
/// the result to all followers. If the leader is dropped without calling
/// [`complete`], an error is automatically broadcast so followers are not
/// left waiting indefinitely.
pub struct TongZhu {
    key: String,
    tx: broadcast::Sender<Result<Vec<u8>>>,
    map: Arc<DashMap<String, broadcast::Sender<Result<Vec<u8>>>>>,
    completed: AtomicBool,
}

impl TongZhu {
    /// Broadcast the result of the operation to all followers.
    ///
    /// This is idempotent: only the first call has any effect; subsequent
    /// calls are silently ignored.
    pub fn complete(&self, result: Result<Vec<u8>>) {
        if !self.completed.swap(true, Ordering::SeqCst) {
            let n = self.tx.send(result).unwrap_or(0);
            debug!(
                "dedupe leader broadcast result for '{}' to {} followers",
                self.key, n
            );
        }
    }

    /// Return the deduplication key for this leader.
    pub fn key(&self) -> &str {
        &self.key
    }
}

impl Drop for TongZhu {
    fn drop(&mut self) {
        if !self.completed.load(Ordering::SeqCst) {
            let err = FeiMingError::Unknown(format!(
                "dedupe leader for '{}' dropped without completion",
                self.key
            ));
            let _ = self.tx.send(Err(err));
            warn!(
                "dedupe leader for '{}' dropped without completion",
                self.key
            );
        }
        self.map.remove(&self.key);
    }
}

/// Singleflight-style request deduplication guard.
///
/// Multiple concurrent callers with the same key will be coalesced: one
/// becomes the "leader" and executes the operation, while the rest become
/// "followers" that wait for the leader's result via a [`broadcast`] channel.
pub struct ShangTong {
    inflight: Arc<DashMap<String, broadcast::Sender<Result<Vec<u8>>>>>,
}

impl ShangTong {
    /// Create a new deduplication guard with an empty in-flight map.
    pub fn new() -> Self {
        Self {
            inflight: Arc::new(DashMap::new()),
        }
    }

    /// Attempt to become the leader for `key`, or register as a follower.
    ///
    /// # Returns
    /// * `TongJu::Leader` — caller must execute the operation and call
    ///   [`TongZhu::complete`] with the result.
    /// * `TongJu::Follower` — caller should `.await` the receiver for
    ///   the leader's result.
    pub fn try_leader(&self, key: &str) -> TongJu {
        // Fast path: key already in flight.
        if let Some(entry) = self.inflight.get(key) {
            let tx = entry.value().clone();
            drop(entry);
            let rx = tx.subscribe();
            debug!("dedupe follower for '{}'", key);
            return TongJu::Follower(rx);
        }

        // Slow path: create a broadcast channel and attempt insertion.
        let (tx, _rx) = broadcast::channel::<Result<Vec<u8>>>(128);
        match self.inflight.entry(key.to_string()) {
            dashmap::mapref::entry::Entry::Occupied(occupied) => {
                let existing_tx = occupied.get().clone();
                let rx = existing_tx.subscribe();
                debug!("dedupe follower (raced) for '{}'", key);
                TongJu::Follower(rx)
            }
            dashmap::mapref::entry::Entry::Vacant(vacant) => {
                vacant.insert(tx.clone());
                debug!("dedupe leader for '{}'", key);
                TongJu::Leader(TongZhu {
                    key: key.to_string(),
                    tx,
                    map: self.inflight.clone(),
                    completed: AtomicBool::new(false),
                })
            }
        }
    }

    /// Return the number of currently in-flight deduplicated operations.
    pub fn inflight_count(&self) -> usize {
        self.inflight.len()
    }
}

impl Default for ShangTong {
    fn default() -> Self {
        Self::new()
    }
}

/// Simple async semaphore-based rate limiter.
pub struct JieYong {
    semaphore: Semaphore,
}

impl JieYong {
    /// Create a new rate limiter allowing up to `max_concurrent` concurrent operations.
    pub fn new(max_concurrent: usize) -> Self {
        Self {
            semaphore: Semaphore::new(max_concurrent),
        }
    }

    /// Acquire a permit, waiting asynchronously if necessary.
    ///
    /// # Panics
    /// Panics if the semaphore has been closed (this never happens in normal operation).
    pub async fn acquire(&self) -> tokio::sync::SemaphorePermit<'_> {
        self.semaphore.acquire().await.expect("semaphore never closed")
    }

    /// Try to acquire a permit immediately without waiting.
    ///
    /// Returns `Err(FeiMingError::RateLimited)` if no permit is available.
    pub fn try_acquire(&self) -> Result<tokio::sync::SemaphorePermit<'_>> {
        self.semaphore
            .try_acquire()
            .map_err(|_| FeiMingError::RateLimited)
    }

    /// Return the number of currently available permits.
    pub fn available_permits(&self) -> usize {
        self.semaphore.available_permits()
    }
}

/// A job to preload a model into memory with a given priority.
///
/// Lower `priority` values are processed first (min-heap ordering).
#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
pub struct HuLing {
    /// Unique identifier for the model.
    pub model_id: String,
    /// Filesystem path to the model weights.
    pub model_path: String,
    /// Priority level — lower values are higher priority.
    pub priority: i32,
}

/// Background preloader that processes [`HuLing`] items with priority ordering
/// and bounded concurrency.
///
/// Jobs are submitted via [`BaiHu::preload`] or [`BaiHu::try_preload`]
/// and processed by a background Tokio task that maintains a min-heap ordered
/// by priority and respects a concurrency semaphore.
pub struct BaiHu {
    tx: mpsc::Sender<HuLing>,
}

impl BaiHu {
    /// Create and start a background preloader.
    ///
    /// # Arguments
    /// * `queue_size` — maximum number of pending jobs in the MPSC channel.
    /// * `concurrency` — maximum number of concurrent preload operations.
    /// * `processor` — async callback invoked for each job. It receives the job
    ///   and a `SemaphorePermit` that is held for the duration of processing.
    pub fn new<P, F>(queue_size: usize, concurrency: usize, processor: P) -> Self
    where
        P: Fn(HuLing, tokio::sync::OwnedSemaphorePermit) -> F + Send + Sync + 'static,
        F: Future<Output = ()> + Send + 'static,
    {
        let (tx, mut rx) = mpsc::channel::<HuLing>(queue_size);
        let concurrency_semaphore = Arc::new(Semaphore::new(concurrency));

        tokio::spawn(async move {
            // Priority queue: lower priority value = higher priority (processed first).
            let mut heap: BinaryHeap<Reverse<(i32, HuLing)>> = BinaryHeap::new();
            let mut join_set: JoinSet<()> = JoinSet::new();

            loop {
                // Try to receive the next job with a 200 ms timeout.
                let recv_result = timeout(Duration::from_millis(200), rx.recv()).await;

                match recv_result {
                    Ok(Some(job)) => {
                        heap.push(Reverse((job.priority, job)));
                    }
                    Ok(None) => {
                        // Channel closed — drain remaining jobs and then exit.
                        debug!("preload channel closed; draining remaining {} jobs", heap.len());
                        break;
                    }
                    Err(_) => {
                        // Timeout — nothing received in 200ms; loop to drain heap below.
                    }
                }

                // Process as many jobs as permits allow.
                while let Some(Reverse((_, job))) = heap.peek() {
                    match concurrency_semaphore.clone().try_acquire_owned() {
                        Ok(permit) => {
                            let popped = heap.pop().unwrap();
                            let fut = processor(popped.0.1, permit);
                            join_set.spawn(fut);
                        }
                        Err(_) => {
                            // At concurrency limit — stop draining for now.
                            break;
                        }
                    }
                }

                // Clean up completed tasks from the join set (non-blocking).
                while let Ok(Some(_)) =
                    tokio::time::timeout(Duration::from_nanos(1), join_set.join_next()).await
                {
                    // Task completed; loop to clean up more.
                }
            }

            // Final drain: process all remaining jobs in the heap.
            while let Some(Reverse((_, job))) = heap.pop() {
                if let Ok(permit) = concurrency_semaphore.clone().try_acquire_owned() {
                    let fut = processor(job, permit);
                    join_set.spawn(fut);
                } else {
                    // Wait for a permit to become available.
                    let permit = concurrency_semaphore
                        .clone()
                        .acquire_owned()
                        .await
                        .expect("semaphore never closed");
                    let fut = processor(job, permit);
                    join_set.spawn(fut);
                }
            }

            // Wait for all in-flight tasks to complete before shutting down.
            while join_set.join_next().await.is_some() {}
            debug!("preload background task shut down cleanly");
        });

        Self { tx }
    }

    /// Send a preload job asynchronously, waiting if the queue is full.
    pub async fn preload(&self, job: HuLing) -> Result<()> {
        self.tx
            .send(job)
            .await
            .map_err(|_| FeiMingError::Unknown("preload channel closed".to_string()))
    }

    /// Try to send a preload job without waiting.
    ///
    /// Returns `Err(FeiMingError::Unknown)` if the queue is full.
    pub fn try_preload(&self, job: HuLing) -> Result<()> {
        self.tx
            .try_send(job)
            .map_err(|_| FeiMingError::Unknown("preload queue full".to_string()))
    }

    /// Clone the internal sender for external use.
    pub fn sender(&self) -> mpsc::Sender<HuLing> {
        self.tx.clone()
    }
}

/// Compute an exponential-backoff duration with jitter for the given attempt.
///
/// The formula is `min(base_delay_ms * 2^attempt, 30_000) + random_jitter(0..1000)`.
/// The shift is capped at 62 to prevent overflow.
fn compute_backoff(attempt: u32, base_delay_ms: u64) -> Duration {
    let shift = attempt.min(62);
    let exponential = base_delay_ms.saturating_mul(1u64 << shift);
    let capped = exponential.min(30_000);
    let jitter = (SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap_or_default()
        .as_nanos()
        % 1000) as u64;
    Duration::from_millis(capped.saturating_add(jitter))
}

/// Execute an async operation with configurable retry and exponential backoff.
///
/// # Arguments
/// * `max_retries` — maximum number of retry attempts (total attempts = max_retries + 1).
/// * `base_delay_ms` — base delay in milliseconds for exponential backoff.
/// * `per_attempt_timeout` — timeout duration for each individual attempt.
/// * `cb` — optional circuit breaker; on success it records a success, on failure
///   it records a failure.
/// * `factory` — closure that produces a future for each attempt.
///
/// # Returns
/// The first successful result, or the last error encountered after all retries.
pub async fn retry_with_backoff<F, Fut, T>(
    max_retries: u32,
    base_delay_ms: u64,
    per_attempt_timeout: Duration,
    cb: Option<&FeiGong>,
    factory: F,
) -> Result<T>
where
    F: Fn() -> Fut,
    Fut: Future<Output = Result<T>>,
{
    let mut last_error: Option<FeiMingError> = None;

    for attempt in 0..=max_retries {
        // Check circuit breaker before attempting.
        if let Some(cb) = cb {
            if let Err(e) = cb.allow() {
                return Err(e);
            }
        }

        match timeout(per_attempt_timeout, factory()).await {
            Ok(Ok(val)) => {
                // Success — record on CB and return.
                if let Some(cb) = cb {
                    cb.record_success();
                }
                return Ok(val);
            }
            Ok(Err(e)) => {
                // Operation returned an error.
                last_error = Some(e);
                if let Some(cb) = cb {
                    cb.record_failure();
                }
            }
            Err(_elapsed) => {
                // Timeout.
                last_error = Some(FeiMingError::Timeout);
                if let Some(cb) = cb {
                    cb.record_failure();
                }
            }
        }

        // If we have retries remaining, compute backoff and sleep.
        if attempt < max_retries {
            let backoff = compute_backoff(attempt, base_delay_ms);
            debug!(
                "retry attempt {}/{} failed, sleeping for {:?}",
                attempt + 1,
                max_retries + 1,
                backoff
            );
            sleep(backoff).await;
        }
    }

    // All retries exhausted — return the last error.
    Err(last_error.unwrap_or_else(|| FeiMingError::Unknown("all retries exhausted with no error".to_string())))
}

/// Composed resilience stack combining circuit breaker, deduplication, and rate limiting.
///
/// [`JianAi::execute`] orchestrates all three layers:
/// 1. Acquire rate-limit permit.
/// 2. Check circuit breaker.
/// 3. Attempt deduplication (leader executes, followers wait).
/// 4. Execute the operation with retry and backoff.
/// 5. Broadcast the result to followers (if leader).
pub struct JianAi {
    /// Circuit breaker for the stack.
    pub cb: Arc<FeiGong>,
    /// Deduplication guard for the stack.
    pub dedupe: Arc<ShangTong>,
    /// Rate limiter for the stack.
    pub limiter: Arc<JieYong>,
}

impl JianAi {
    /// Create a new resilience stack from its constituent parts.
    pub fn new(
        cb: Arc<FeiGong>,
        dedupe: Arc<ShangTong>,
        limiter: Arc<JieYong>,
    ) -> Self {
        Self { cb, dedupe, limiter }
    }

    /// Execute an operation through the full resilience stack.
    ///
    /// The operation goes through rate limiting, circuit breaker gating,
    /// deduplication, and retry with exponential backoff.
    ///
    /// # Arguments
    /// * `dedupe_key` — key used for deduplication; concurrent callers with
    ///   the same key will share one execution.
    /// * `operation_name` — human-readable name for logging.
    /// * `max_retries` — maximum retry attempts.
    /// * `per_attempt_timeout` — timeout for each attempt.
    /// * `factory` — closure producing the async operation future.
    ///
    /// # Returns
    /// The operation result (shared among all deduplicated callers).
    pub async fn execute<F, Fut>(
        &self,
        dedupe_key: &str,
        operation_name: &str,
        max_retries: u32,
        per_attempt_timeout: Duration,
        factory: F,
    ) -> Result<Vec<u8>>
    where
        F: Fn() -> Fut + Send + Sync + 'static,
        Fut: Future<Output = Result<Vec<u8>>> + Send + 'static,
    {
        // 1. Rate limit.
        let _permit = self.limiter.acquire().await;
        debug!("rate limit acquired for '{}'", operation_name);

        // 2. Circuit breaker check.
        self.cb.allow()?;

        // 3. Deduplication.
        match self.dedupe.try_leader(dedupe_key) {
            TongJu::Follower(mut rx) => {
                debug!(
                    "dedupe follower waiting for '{}' on key '{}'",
                    operation_name, dedupe_key
                );
                match rx.recv().await {
                    Ok(result) => {
                        debug!("dedupe follower received result for '{}'", operation_name);
                        result
                    }
                    Err(broadcast::error::RecvError::Closed) => {
                        error!("dedupe broadcast closed");
                        Err(FeiMingError::Unknown(
                            "dedupe broadcast channel closed".to_string(),
                        ))
                    }
                    Err(broadcast::error::RecvError::Lagged(n)) => {
                        warn!("dedupe follower lagged by {} messages", n);
                        // Try to receive once more after lagging.
                        match rx.recv().await {
                            Ok(r) => r,
                            Err(_) => Err(FeiMingError::Unknown(
                                "dedupe broadcast lagged and closed".to_string(),
                            )),
                        }
                    }
                }
            }
            TongJu::Leader(leader) => {
                debug!(
                    "dedupe leader executing '{}' on key '{}'",
                    operation_name, dedupe_key
                );

                // Execute with retry and backoff.
                let result = retry_with_backoff(
                    max_retries,
                    100,
                    per_attempt_timeout,
                    Some(&*self.cb),
                    &factory,
                )
                .await;

                // Broadcast the result to all followers.
                leader.complete(result.clone());
                result
            }
        }
    }
}

    /// Renders all registered metrics in Prometheus text exposition format.
    ///
    /// Returns a UTF-8 string containing all metric families and their
    /// current values, suitable for scraping by Prometheus.
    pub fn render(&self) -> String {
        let encoder = TextEncoder::new();
        let metric_families = self.registry.gather();
        let mut buf = Vec::new();
        // encode returns a Result; on error we return an empty string
        if encoder.encode(&metric_families, &mut buf).is_err() {
            return String::new();
        }
        String::from_utf8(buf).unwrap_or_default()
    }

/// Overall health status of the 非攻兼爱系统.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
pub enum ChaJi {
    /// All checks pass; system is operating normally.
    Healthy,
    /// One or more checks report warnings but no failures.
    Degraded,
    /// One or more checks report failures.
    Unhealthy,
}

impl fmt::Display for ChaJi {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            ChaJi::Healthy => write!(f, "生机旺盛"),
            ChaJi::Degraded => write!(f, "生机受损"),
            ChaJi::Unhealthy => write!(f, "生机告危"),
        }
    }
}

/// Status of an individual component health check.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
pub enum YanJi {
    /// The component check passed.
    Pass,
    /// The component check reported a warning.
    Warn,
    /// The component check reported a failure.
    Fail,
}

impl fmt::Display for YanJi {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            YanJi::Pass => write!(f, "通达"),
            YanJi::Warn => write!(f, "预警"),
            YanJi::Fail => write!(f, "告破"),
        }
    }
}

/// Result of a single component health check.
#[derive(Debug, Clone, Serialize)]
pub struct ZuJianCha {
    /// Name of the component being checked.
    pub name: String,
    /// Status of this component check.
    pub status: YanJi,
    /// Human-readable message describing the check result.
    pub message: String,
    /// Latency of the check in milliseconds.
    pub latency_ms: u64,
}

/// Complete health report aggregating all component checks.
#[derive(Debug, Clone, Serialize)]
pub struct MingGuiBao {
    /// Aggregated health status across all checks.
    pub status: ChaJi,
    /// Individual component check results.
    pub checks: Vec<ZuJianCha>,
    /// Unix timestamp (seconds) when the report was generated.
    pub timestamp: u64,
}

/// Performs health checks against 非攻兼爱系统各关键子系统.
///
/// Evaluates circuit breaker state, cache tiers, memory pressure,
/// inference errors, backend RPC health, and model loading status to produce
/// a comprehensive `MingGuiBao`.
pub struct MingGui {
    cb: Arc<FeiGong>,
    cache: Arc<JiGuanCheng>,
    metrics: Arc<TianZhi>,
    rss_quota_bytes: u64,
}

impl MingGui {
    /// Creates a new `MingGui`.
    ///
    /// # Arguments
    /// * `cb`               - The circuit breaker to monitor.
    /// * `cache`            - The tiered cache to inspect.
    /// * `metrics`          - The metrics registry to read current values from.
    /// * `rss_quota_bytes`  - The memory quota limit in bytes.
    pub fn new(
        cb: Arc<FeiGong>,
        cache: Arc<JiGuanCheng>,
        metrics: Arc<TianZhi>,
        rss_quota_bytes: u64,
    ) -> Self {
        MingGui {
            cb,
            cache,
            metrics,
            rss_quota_bytes,
        }
    }

    /// Returns `true` if the process is alive (liveness probe).
    ///
    /// Always returns `true` — if this code is executing, the process is running.
    pub fn check_liveness(&self) -> bool {
        true
    }

    /// Returns `true` if the system is ready to serve traffic (readiness probe).
    ///
    /// Returns `true` when the overall health status is `Healthy` or `Degraded`.
    /// Returns `false` only when the status is `Unhealthy`.
    pub fn check_readiness(&self) -> bool {
        let report = self.check_health();
        report.status == ChaJi::Healthy || report.status == ChaJi::Degraded
    }

    /// Performs a comprehensive health check across all subsystems.
    ///
    /// Evaluates 8 component checks and aggregates them into a `MingGuiBao`.
    pub fn check_health(&self) -> MingGuiBao {
        let start = SystemTime::now();
        let mut checks = Vec::with_capacity(8);

        // --- 1. Circuit breaker check ---
        let cb_state = self.cb.state_u32();
        let cb_status = match cb_state {
            2 => YanJi::Fail,
            1 => YanJi::Warn,
            _ => YanJi::Pass,
        };
        checks.push(ZuJianCha {
            name: "机关锁".to_string(),
            status: cb_status,
            message: format!("CB state={} (0=closed,1=half,2=open)", cb_state),
            latency_ms: 0,
        });

        // --- 2. L1 cache check ---
        let l1_hits = self.metrics.l1_hit_total.get() as u64;
        let l1_misses = self.metrics.l1_miss_total.get() as u64;
        let l1_entries = self.metrics.l1_entries_gauge.get() as u64;
        let l1_status = if l1_entries == 0 {
            YanJi::Warn
        } else {
            YanJi::Pass
        };
        checks.push(ZuJianCha {
            name: "速印台".to_string(),
            status: l1_status,
            message: format!(
                "L1 entries={}, hits={}, misses={}",
                l1_entries, l1_hits, l1_misses
            ),
            latency_ms: 0,
        });

        // --- 3. L2 cache check ---
        let l2_hits = self.metrics.l2_hit_total.get() as u64;
        let l2_misses = self.metrics.l2_miss_total.get() as u64;
        let l2_size = self.metrics.l2_size_bytes.get() as u64;
        let l2_entries_approx = l2_size.saturating_div(4096).max(1);
        let l2_status = if l2_size == 0 {
            YanJi::Warn
        } else {
            YanJi::Pass
        };
        checks.push(ZuJianCha {
            name: "炉印台".to_string(),
            status: l2_status,
            message: format!(
                "L2 entries≈{}, size={}B, hits={}, misses={}",
                l2_entries_approx, l2_size, l2_hits, l2_misses
            ),
            latency_ms: 0,
        });

        // --- 4. L3 cache check ---
        let l3_hits = self.metrics.l3_hit_total.get() as u64;
        let l3_misses = self.metrics.l3_miss_total.get() as u64;
        checks.push(ZuJianCha {
            name: "云印台".to_string(),
            status: YanJi::Pass,
            message: format!(
                "L3 hits={}, misses={} (L3 is optional, always pass)",
                l3_hits, l3_misses
            ),
            latency_ms: 0,
        });

        // --- 5. Memory pressure check ---
        let rss = self.metrics.rss_bytes.get() as u64;
        let rss_ratio = if self.rss_quota_bytes > 0 {
            rss as f64 / self.rss_quota_bytes as f64
        } else {
            0.0
        };
        let mem_status = if rss_ratio > 0.95 {
            YanJi::Fail
        } else if rss_ratio > 0.80 {
            YanJi::Warn
        } else {
            YanJi::Pass
        };
        checks.push(ZuJianCha {
            name: "内存压力".to_string(),
            status: mem_status,
            message: format!(
                "RSS={}B / quota={}B (ratio={:.2}%)",
                rss, self.rss_quota_bytes, rss_ratio * 100.0
            ),
            latency_ms: 0,
        });

        // --- 6. Inference errors check ---
        let inf_total = self.metrics.inference_total.get();
        let inf_errors = self.metrics.inference_errors_total.get();
        let err_rate = if inf_total > 0.0 {
            inf_errors as f64 / inf_total as f64
        } else {
            0.0
        };
        let err_status = if err_rate > 0.20 {
            YanJi::Fail
        } else if err_rate > 0.05 {
            YanJi::Warn
        } else {
            YanJi::Pass
        };
        checks.push(ZuJianCha {
            name: "推演错误率".to_string(),
            status: err_status,
            message: format!(
                "inference_errors={:.0} / total={:.0} (rate={:.2}%)",
                inf_errors, inf_total, err_rate * 100.0
            ),
            latency_ms: 0,
        });

        // --- 7. Backend RPC check ---
        let rpc_total = self.metrics.backend_rpc_total.get();
        let rpc_errors = self.metrics.backend_rpc_errors_total.get();
        let rpc_err_rate = if rpc_total > 0.0 {
            rpc_errors as f64 / rpc_total as f64
        } else {
            0.0
        };
        let rpc_status = if rpc_err_rate > 0.30 {
            YanJi::Fail
        } else if rpc_err_rate > 0.10 {
            YanJi::Warn
        } else {
            YanJi::Pass
        };
        checks.push(ZuJianCha {
            name: "后端通讯".to_string(),
            status: rpc_status,
            message: format!(
                "rpc_errors={:.0} / total={:.0} (rate={:.2}%)",
                rpc_errors, rpc_total, rpc_err_rate * 100.0
            ),
            latency_ms: 0,
        });

        // --- 8. Model loading check ---
        let loaded = self.metrics.model_loaded_gauge.get() as i64;
        let load_errors = self.metrics.model_load_errors_total.get() as i64;
        let model_status = if load_errors > 0 && loaded == 0 {
            YanJi::Fail
        } else if load_errors > 0 {
            YanJi::Warn
        } else {
            YanJi::Pass
        };
        checks.push(ZuJianCha {
            name: "墨器加载".to_string(),
            status: model_status,
            message: format!(
                "models_loaded={}, load_errors={}",
                loaded, load_errors
            ),
            latency_ms: 0,
        });

        // --- Compute overall status ---
        let overall_status = if checks.iter().any(|c| c.status == YanJi::Fail) {
            ChaJi::Unhealthy
        } else if checks.iter().any(|c| c.status == YanJi::Warn) {
            ChaJi::Degraded
        } else {
            ChaJi::Healthy
        };

        let elapsed_ms = start
            .elapsed()
            .unwrap_or(Duration::ZERO)
            .as_millis() as u64;
        for c in checks.iter_mut() {
            c.latency_ms = elapsed_ms;
        }

        let timestamp = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap_or(Duration::ZERO)
            .as_secs();

        MingGuiBao {
            status: overall_status,
            checks,
            timestamp,
        }
    }
}

/// JiShi level of an operational recommendation.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
pub enum JiShi {
    /// Informational; no action required.
    Info,
    /// Warning; attention recommended.
    Warning,
    /// Critical; immediate action required.
    Critical,
}

impl fmt::Display for JiShi {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            JiShi::Info => write!(f, "通报"),
            JiShi::Warning => write!(f, "警示"),
            JiShi::Critical => write!(f, "紧急"),
        }
    }
}

/// An operational recommendation produced by the operations engine.
#[derive(Debug, Clone, Serialize)]
pub struct ShangXianYi {
    /// JiShi of the recommendation.
    pub severity: JiShi,
    /// Category this recommendation belongs to (e.g. "cache", "latency").
    pub category: String,
    /// Human-readable description of the situation.
    pub message: String,
    /// Suggested remedial action.
    pub suggested_action: String,
}

/// A snapshot of system metrics used for operations analysis.
#[derive(Debug, Clone, Serialize)]
pub struct TianMu {
    /// L1 cache hit rate (0.0 to 1.0).
    pub l1_hit_rate: f64,
    /// L2 cache hit rate (0.0 to 1.0).
    pub l2_hit_rate: f64,
    /// P99 inference latency in milliseconds.
    pub p99_latency_ms: f64,
    /// Inference error rate (0.0 to 1.0).
    pub error_rate: f64,
    /// RSS memory ratio relative to quota (0.0 to 1.0+).
    pub rss_ratio: f64,
    /// Circuit breaker state (0=closed, 1=half-open, 2=open).
    pub cb_state: u32,
    /// Ratio of in-flight requests relative to capacity (0.0 to 1.0+).
    pub inflight_ratio: f64,
}

/// Analyzes system metrics and produces actionable operational recommendations.
///
/// Evaluates 7 individual decision nodes plus 4 combination scenarios to
/// detect common failure patterns and suggest remedial actions.
pub struct ShangXian;

impl ShangXian {
    /// Analyzes a metrics snapshot and returns a list of recommendations.
    ///
    /// Evaluates 7 individual decision nodes covering cache, latency, errors,
    /// memory, circuit breaker, and concurrency. Then checks 4 combination
    /// scenarios for compound failure patterns. If no issues are detected,
    /// returns a single "all systems nominal" info recommendation.
    ///
    /// # Arguments
    /// * `snapshot` - A `TianMu` containing current system metrics.
    pub fn analyze_situation(snapshot: &TianMu) -> Vec<ShangXianYi> {
        let mut recs: Vec<ShangXianYi> = Vec::new();

        // --- Decision Node 1: L1 hit rate ---
        if snapshot.l1_hit_rate < 0.5 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "cache".to_string(),
                message: format!(
                    "L1 cache hit rate is low ({:.1}%)",
                    snapshot.l1_hit_rate * 100.0
                ),
                suggested_action: "Increase L1 capacity or reduce TTL".to_string(),
            });
        }

        // --- Decision Node 2: L2 hit rate ---
        if snapshot.l2_hit_rate < 0.3 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "cache".to_string(),
                message: format!(
                    "L2 cache hit rate is low ({:.1}%)",
                    snapshot.l2_hit_rate * 100.0
                ),
                suggested_action: "Increase L2 disk budget or warm cache".to_string(),
            });
        }

        // --- Decision Node 3: P99 latency ---
        if snapshot.p99_latency_ms > 2000.0 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "latency".to_string(),
                message: format!(
                    "P99 inference latency is critically high ({:.0}ms)",
                    snapshot.p99_latency_ms
                ),
                suggested_action: "Scale replicas, reduce model load time".to_string(),
            });
        } else if snapshot.p99_latency_ms > 500.0 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "latency".to_string(),
                message: format!(
                    "P99 inference latency is elevated ({:.0}ms)",
                    snapshot.p99_latency_ms
                ),
                suggested_action: "Scale replicas, reduce model load time".to_string(),
            });
        }

        // --- Decision Node 4: Error rate ---
        if snapshot.error_rate > 0.20 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "errors".to_string(),
                message: format!(
                    "Inference error rate is critically high ({:.1}%)",
                    snapshot.error_rate * 100.0
                ),
                suggested_action: "Check backend health, review model compatibility".to_string(),
            });
        } else if snapshot.error_rate > 0.05 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "errors".to_string(),
                message: format!(
                    "Inference error rate is elevated ({:.1}%)",
                    snapshot.error_rate * 100.0
                ),
                suggested_action: "Check backend health, review model compatibility".to_string(),
            });
        }

        // --- Decision Node 5: RSS memory ratio ---
        if snapshot.rss_ratio > 0.95 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "memory".to_string(),
                message: format!(
                    "Memory usage is critically high ({:.1}% of quota)",
                    snapshot.rss_ratio * 100.0
                ),
                suggested_action: "Immediately evict cold models".to_string(),
            });
        } else if snapshot.rss_ratio > 0.80 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "memory".to_string(),
                message: format!(
                    "Memory usage is elevated ({:.1}% of quota)",
                    snapshot.rss_ratio * 100.0
                ),
                suggested_action: "Evict cold models or increase memory quota".to_string(),
            });
        }

        // --- Decision Node 6: Circuit breaker state ---
        if snapshot.cb_state == 2 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "resilience".to_string(),
                message: "Circuit breaker is OPEN".to_string(),
                suggested_action: "Investigate backend failures, reset circuit breaker manually".to_string(),
            });
        } else if snapshot.cb_state == 1 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "resilience".to_string(),
                message: "Circuit breaker is HALF-OPEN".to_string(),
                suggested_action: "Monitor closely; backend may be recovering".to_string(),
            });
        }

        // --- Decision Node 7: Inflight ratio ---
        if snapshot.inflight_ratio > 0.90 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "concurrency".to_string(),
                message: format!(
                    "Inflight request ratio is critically high ({:.1}%)",
                    snapshot.inflight_ratio * 100.0
                ),
                suggested_action: "Scale replicas or shed load".to_string(),
            });
        } else if snapshot.inflight_ratio > 0.70 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "concurrency".to_string(),
                message: format!(
                    "Inflight request ratio is elevated ({:.1}%)",
                    snapshot.inflight_ratio * 100.0
                ),
                suggested_action: "Scale replicas or shed load".to_string(),
            });
        }

        // --- Combination Scenario 1: L1 < 0.3 AND P99 > 1000 ---
        if snapshot.l1_hit_rate < 0.3 && snapshot.p99_latency_ms > 1000.0 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "combo".to_string(),
                message: "Cache thrashing: low L1 + high latency".to_string(),
                suggested_action: "Increase L1 capacity and investigate cache eviction patterns".to_string(),
            });
        }

        // --- Combination Scenario 2: Error rate > 15% AND CB == 2 ---
        if snapshot.error_rate > 0.15 && snapshot.cb_state == 2 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "combo".to_string(),
                message: "Cascading failure: high errors + circuit breaker open".to_string(),
                suggested_action: "Emergency: all backends may be down, engage incident response".to_string(),
            });
        }

        // --- Combination Scenario 3: RSS > 85% AND L2 < 0.2 ---
        if snapshot.rss_ratio > 0.85 && snapshot.l2_hit_rate < 0.2 {
            recs.push(ShangXianYi {
                severity: JiShi::Warning,
                category: "combo".to_string(),
                message: "Memory and disk cache both under pressure".to_string(),
                suggested_action: "Increase L2 budget, evict cold models from memory".to_string(),
            });
        }

        // --- Combination Scenario 4: Inflight > 80% AND P99 > 1500 ---
        if snapshot.inflight_ratio > 0.80 && snapshot.p99_latency_ms > 1500.0 {
            recs.push(ShangXianYi {
                severity: JiShi::Critical,
                category: "combo".to_string(),
                message: "Overload detected: high concurrency + high latency".to_string(),
                suggested_action: "Enable load shedding, scale out immediately".to_string(),
            });
        }

        // --- Default: all systems nominal ---
        if recs.is_empty() {
            recs.push(ShangXianYi {
                severity: JiShi::Info,
                category: "general".to_string(),
                message: "All systems nominal".to_string(),
                suggested_action: "No action required".to_string(),
            });
        }

        recs
    }
}

/// Request payload for loading a model into the inference service.
///
/// The `key` uniquely identifies the model, `payload` contains the serialized
/// model bytes, and `meta` / `kind` provide optional metadata and model-type
/// hints.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ZaiQiQiu {
    /// Unique model identifier (e.g. `"resnet50::v1"`).
    pub key: String,
    /// Serialized model bytes (e.g. ONNX blob, TorchScript, etc.).
    pub payload: Vec<u8>,
    /// Optional JSON metadata string.
    pub meta: Option<String>,
    /// Optional model-kind discriminator (e.g. `"onnx"`, `"torch"`).
    pub kind: Option<String>,
    /// Maximum time allowed for the load operation, in milliseconds.
    pub timeout_ms: u64,
}

/// Response returned after a successful model load.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ZaiQiYing {
    /// Opaque handle ID assigned by the loader.  All subsequent inference
    /// requests for this model must include this handle.
    pub handle_id: String,
    /// JSON-encoded metadata returned by the loader.
    pub meta_json: String,
}

/// Request payload for unloading (evicting) a previously loaded model.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct XieQiQiu {
    /// Handle ID returned by the corresponding `ZaiQiYing`.
    pub handle_id: String,
}

/// Response returned after an unload request.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct XieQiYing {
    /// `true` if the model was found and removed.
    pub ok: bool,
}

/// Request payload for a single inference call.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TuiYanQiu {
    /// Handle ID of the loaded model to run inference against.
    pub handle_id: String,
    /// Serialized input tensor / feature vector.
    pub input: Vec<u8>,
}

/// Response returned after a successful inference call.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TuiYanYing {
    /// Serialized output tensor / prediction vector.
    pub output: Vec<u8>,
}

/// Response returned by the loader health-check endpoint.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ShengJiYing {
    /// Human-readable status string (e.g. `"ok"`, `"degraded"`).
    pub status: String,
}

/// Response returned by the loader handle-listing endpoint.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ZhaMingYing {
    /// All currently active model handle IDs.
    pub handles: Vec<String>,
}

/// Cross-platform RSS (Resident Set Size) memory monitor.
///
/// On Linux `rss_bytes` reads `/proc/self/statm` and multiplies by the system
/// page size for an O(1) query.  On other platforms it falls back to
/// `sysinfo::System` which performs a process-table walk.
pub struct NeiShi {
    /// Cached page size in bytes (obtained once at construction).
    page_size: u64,
}

impl NeiShi {
    /// Create a new `NeiShi` and cache the system page size.
    pub fn new() -> Self {
        let page_size = unsafe { libc::sysconf(libc::_SC_PAGESIZE) as u64 };
        Self { page_size }
    }

    /// Return the RSS of the current process in bytes.
    ///
    /// Uses `/proc/self/statm` on Linux; falls back to `sysinfo` on other
    /// operating systems.
    #[cfg(target_os = "linux")]
    pub fn rss_bytes(&self) -> u64 {
        let content = std::fs::read_to_string("/proc/self/statm").unwrap_or_default();
        let parts: Vec<&str> = content.split_whitespace().collect();
        if parts.len() > 1 {
            if let Ok(rss_pages) = parts[1].parse::<u64>() {
                return rss_pages * self.page_size;
            }
        }
        self.fallback_rss()
    }

    /// Non-Linux placeholder that always delegates to the `sysinfo` fallback.
    #[cfg(not(target_os = "linux"))]
    pub fn rss_bytes(&self) -> u64 {
        self.fallback_rss()
    }

    /// Fallback RSS measurement using `sysinfo::System`.
    ///
    /// Refreshes the full system state, looks up the current process by PID,
    /// and returns its resident memory in bytes.
    fn fallback_rss(&self) -> u64 {
        let mut system = sysinfo::System::new_all();
        system.refresh_all();
        let pid = std::process::id() as usize;
        system
            .process(sysinfo::Pid::from(pid))
            .map(|p| p.memory())
            .unwrap_or(0)
    }
}

impl Default for NeiShi {
    fn default() -> Self {
        Self::new()
    }
}

/// HTTP client for 与非攻兼爱智器加载服务通讯.
///
/// Provides typed methods for loading/unloading models, submitting inference
/// requests, health checks, and listing active handles.
#[derive(Clone)]
pub struct ZhiQi {
    /// Underlying `reqwest` HTTP client (reused for connection pooling).
    client: reqwest::Client,
    /// Base URL of the loader service (e.g. `http://127.0.0.1:8081`).
    base_url: String,
}

impl ZhiQi {
    /// Build a new `ZhiQi` without performing a health check.
    ///
    /// The internal `reqwest::Client` is configured with:
    /// - `tcp_nodelay(true)` — disable Nagle for low-latency RPC.
    /// - 5-second connect timeout.
    /// - 30-second overall request timeout.
    pub fn new(base_url: String) -> Result<Self> {
        let client = reqwest::Client::builder()
            .tcp_nodelay(true)
            .connect_timeout(std::time::Duration::from_secs(5))
            .timeout(std::time::Duration::from_secs(30))
            .build()
            .map_err(|e| {
                FeiMingError::Network(format!("failed to build HTTP client: {}", e))
            })?;
        Ok(Self { client, base_url })
    }

    /// Build a new `ZhiQi` and verify connectivity with a health
    /// check.
    ///
    /// The health check uses a 5-second timeout.  If the loader does not
    /// respond with HTTP 200 and a `"status"` field, an error is returned.
    pub async fn connect(base_url: &str) -> Result<Self> {
        let client = reqwest::Client::builder()
            .tcp_nodelay(true)
            .connect_timeout(std::time::Duration::from_secs(5))
            .timeout(std::time::Duration::from_secs(30))
            .build()
            .map_err(|e| {
                FeiMingError::Network(format!("failed to build HTTP client: {}", e))
            })?;

        let health_url = format!("{}/health", base_url);
        let resp = client
            .get(&health_url)
            .timeout(std::time::Duration::from_secs(5))
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!("health check request failed: {}", e))
            })?;

        if !resp.status().is_success() {
            return Err(FeiMingError::Network(format!(
                "health check returned non-success status: {}",
                resp.status()
            )));
        }

        let health: ShengJiYing = resp.json().await.map_err(|e| {
            FeiMingError::Serialization(format!("failed to parse health response: {}", e))
        })?;

        if health.status != "ok" && health.status != "healthy" {
            return Err(FeiMingError::Network(format!(
                "loader health status is not ok: {}",
                health.status
            )));
        }

        Ok(Self {
            client,
            base_url: base_url.to_string(),
        })
    }

    /// Load a model into the loader service.
    ///
    /// Returns `(handle_id, meta_json)` on success.  The request timeout is
    /// derived from `timeout_ms`.
    pub async fn load_model(
        &self,
        key: String,
        payload: Vec<u8>,
        meta: Option<String>,
        kind: Option<String>,
        timeout_ms: u64,
    ) -> Result<(String, String)> {
        let req_body = ZaiQiQiu {
            key,
            payload,
            meta,
            kind,
            timeout_ms,
        };

        let url = format!("{}/load", self.base_url);
        let resp = self
            .client
            .post(&url)
            .json(&req_body)
            .timeout(std::time::Duration::from_millis(timeout_ms))
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!("load_model request failed: {}", e))
            })?;

        if !resp.status().is_success() {
            let status = resp.status();
            let body_text = resp.text().await.unwrap_or_default();
            return Err(FeiMingError::Network(format!(
                "load_model returned {}: {}",
                status, body_text
            )));
        }

        let load_resp: ZaiQiYing = resp.json().await.map_err(|e| {
            FeiMingError::Serialization(format!("failed to parse ZaiQiYing: {}", e))
        })?;

        Ok((load_resp.handle_id, load_resp.meta_json))
    }

    /// Unload (evict) a previously loaded model.
    ///
    /// Returns `true` if the model was found and removed.
    pub async fn unload_model(&self, handle_id: String) -> Result<bool> {
        let req_body = XieQiQiu { handle_id };

        let url = format!("{}/unload", self.base_url);
        let resp = self
            .client
            .post(&url)
            .json(&req_body)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!("unload_model request failed: {}", e))
            })?;

        if !resp.status().is_success() {
            let status = resp.status();
            let body_text = resp.text().await.unwrap_or_default();
            return Err(FeiMingError::Network(format!(
                "unload_model returned {}: {}",
                status, body_text
            )));
        }

        let unload_resp: XieQiYing = resp.json().await.map_err(|e| {
            FeiMingError::Serialization(format!("failed to parse XieQiYing: {}", e))
        })?;

        Ok(unload_resp.ok)
    }

    /// Submit an inference request for a loaded model.
    ///
    /// Returns the serialized output tensor bytes on success.
    pub async fn infer(&self, handle_id: String, input: Vec<u8>) -> Result<Vec<u8>> {
        let req_body = TuiYanQiu { handle_id, input };

        let url = format!("{}/infer", self.base_url);
        let resp = self
            .client
            .post(&url)
            .json(&req_body)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!("infer request failed: {}", e))
            })?;

        if !resp.status().is_success() {
            let status = resp.status();
            let body_text = resp.text().await.unwrap_or_default();
            return Err(FeiMingError::Network(format!(
                "infer returned {}: {}",
                status, body_text
            )));
        }

        let infer_resp: TuiYanYing = resp.json().await.map_err(|e| {
            FeiMingError::Serialization(format!("failed to parse TuiYanYing: {}", e))
        })?;

        Ok(infer_resp.output)
    }

    /// Query the loader's health endpoint.
    ///
    /// Returns the human-readable status string (e.g. `"ok"`).
    pub async fn health(&self) -> Result<String> {
        let url = format!("{}/health", self.base_url);
        let resp = self
            .client
            .get(&url)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!("health request failed: {}", e))
            })?;

        if !resp.status().is_success() {
            let status = resp.status();
            let body_text = resp.text().await.unwrap_or_default();
            return Err(FeiMingError::Network(format!(
                "health returned {}: {}",
                status, body_text
            )));
        }

        let health_resp: ShengJiYing = resp.json().await.map_err(|e| {
            FeiMingError::Serialization(format!("failed to parse ShengJiYing: {}", e))
        })?;

        Ok(health_resp.status)
    }

    /// List all currently active model handles.
    ///
    /// Returns a vector of opaque handle ID strings.
    pub async fn list_handles(&self) -> Result<Vec<String>> {
        let url = format!("{}/handles", self.base_url);
        let resp = self
            .client
            .get(&url)
            .send()
            .await
            .map_err(|e| {
                FeiMingError::Network(format!("list_handles request failed: {}", e))
            })?;

        if !resp.status().is_success() {
            let status = resp.status();
            let body_text = resp.text().await.unwrap_or_default();
            return Err(FeiMingError::Network(format!(
                "list_handles returned {}: {}",
                status, body_text
            )));
        }

        let list_resp: ZhaMingYing = resp.json().await.map_err(|e| {
            FeiMingError::Serialization(format!("failed to parse ZhaMingYing: {}", e))
        })?;

        Ok(list_resp.handles)
    }
}

/// Application state shared across all HTTP handlers.
///
/// All fields are wrapped in `Arc` so the state can be cloned cheaply into
/// every Axum handler via `State` extraction.
pub struct ChengTai {
    /// System-wide configuration (log level, cache sizes, endpoints, …).
    pub config: JuZiLing,
    /// Metrics registry for counters, histograms, and gauges.
    pub metrics: Arc<TianZhi>,
    /// Tiered cache (L1 memory → L2 RocksDB → optional L3 S3).
    pub cache: Arc<JiGuanCheng>,
    /// Circuit breaker protecting calls to the loader service.
    pub cb: Arc<FeiGong>,
    /// HTTP client for the model loader.
    pub loader: ZhiQi,
}

/// ---------------------------------------------------------------------------
/// HTTP route handlers
/// ---------------------------------------------------------------------------

/// Health check handler — returns a JSON `MingGuiBao`.
///
/// Collects process RSS, uptime, and subsystem statuses into a single report.
async fn health_handler(State(state): State<Arc<ChengTai>>) -> impl IntoResponse {
    let rss_monitor = NeiShi::new();
    let rss = rss_monitor.rss_bytes();

    let uptime_secs = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap_or_default()
        .as_secs();

    let report = json!({
        "status": "ok",
        "rss_bytes": rss,
        "uptime_secs": uptime_secs,
        "loader_endpoint": state.config.loader_endpoint,
        "version": env!("CARGO_PKG_VERSION"),
    });

    (StatusCode::OK, Json(report))
}

/// Readiness probe — returns HTTP 200 if the service is ready to serve
/// inference requests, otherwise HTTP 503.
///
/// A readiness check verifies that the loader is reachable and the cache
/// tiers are initialised.
async fn ready_handler(State(state): State<Arc<ChengTai>>) -> impl IntoResponse {
    match state.loader.health().await {
        Ok(status) => {
            if status == "ok" || status == "healthy" {
                (StatusCode::OK, "ready")
            } else {
                (StatusCode::SERVICE_UNAVAILABLE, "loader not healthy")
            }
        }
        Err(_) => (StatusCode::SERVICE_UNAVAILABLE, "loader unreachable"),
    }
}

/// Prometheus-compatible metrics endpoint.
///
/// Renders all collected counters, histograms, and gauges as UTF-8 text in
/// the Prometheus exposition format.
async fn metrics_handler(State(state): State<Arc<ChengTai>>) -> impl IntoResponse {
    let body = state.metrics.render();
    (
        StatusCode::OK,
        [(header::CONTENT_TYPE, HeaderValue::from_static("text/plain; charset=utf-8"))],
        body,
    )
}

/// Inference handler — the primary API for submitting inference requests.
///
/// 1. Parse the JSON request body (`TuiYanQingQiu`).
/// 2. Compute a deterministic cache key from the model + input hash.
/// 3. Check the tiered cache — on hit, return the cached output immediately.
/// 4. On cache miss, check the circuit breaker.
/// 5. If the circuit breaker allows, call the loader's `/infer` endpoint.
/// 6. Store the result in the cache and return it to the caller.
async fn infer_handler(
    State(state): State<Arc<ChengTai>>,
    Json(req): Json<TuiYanQingQiu>,
) -> impl IntoResponse {
    // 1. Compute cache key = model_id + ":" + hex_hash_of_inputs
    let input_hash = compute_input_hash(&req.inputs);
    let cache_key = format!("{}:{}", req.model_id, input_hash);

    // 2. Try L1 → L2 → L3 cache
    let cache_result = match state.cache.lookup(&cache_key).await {
        Ok(cr) => cr,
        Err(e) => {
            error!("cache lookup error for key={}: {}", cache_key, e);
            YinYing::miss()
        }
    };

    if let Some(entry) = cache_result.entry {
        debug!("cache hit for key={} from {:?}", cache_key, cache_result.served_from);
        state.metrics.inc_counter("tuiyan_yincun_ming_ci", 1);
        state.metrics.inc_counter("tuiyan_zong_ji", 1);
        let resp = TuiYanHuiFu {
            outputs: vec![], // entry.output_data contains the cached result
            model_id: req.model_id.clone(),
            inference_time_us: entry.inference_time_us,
        };
        return (StatusCode::OK, Json(resp)).into_response();
    }

    debug!("cache miss for key={}", cache_key);
    state.metrics.inc_counter("tuiyan_zong_ji", 1);

    // 3. Circuit breaker check
    if state.cb.allow().is_err() {
        warn!("circuit breaker open — rejecting inference request");
        state.metrics.inc_counter("tuiyan_cuo_wu_ji", 1);
        return (
            StatusCode::SERVICE_UNAVAILABLE,
            Json(json!({"error": "circuit breaker open"})),
        )
            .into_response();
    }

    // 4. Call loader — serialize inputs to Vec<u8> for the loader
    let start = std::time::Instant::now();
    let input_bytes = match bincode_encode(&req.inputs) {
        Ok(b) => b,
        Err(e) => {
            error!("failed to serialize inputs: {}", e);
            return (
                StatusCode::BAD_REQUEST,
                Json(json!({"error": format!("input serialization failed: {}", e)})),
            ).into_response();
        }
    };

    match state.loader.infer(req.model_id.clone(), input_bytes).await {
        Ok(output) => {
            let elapsed = start.elapsed().as_secs_f64();
            state.metrics.observe_histogram("tuiyan_shizhang_miao", elapsed);
            state.metrics.inc_counter("tuiyan_zong_ji", 1);
            state.cb.record_success();

            // 5. Cache the result (best-effort)
            let entry = MoYin::new(
                req.model_id.clone(),
                hash_inputs(&req.inputs),
                output.clone(),
                state.config.l1_ttl_secs,
                start.elapsed().as_micros() as u64,
            );
            if let Err(e) = state.cache.insert(&entry).await {
                warn!("failed to cache inference result for key={}: {}", cache_key, e);
            }

            let resp = TuiYanHuiFu {
                outputs: vec![], // output is raw Vec<u8> from loader
                model_id: req.model_id,
                inference_time_us: start.elapsed().as_micros() as u64,
            };
            (StatusCode::OK, Json(resp)).into_response()
        }
        Err(e) => {
            let elapsed = start.elapsed().as_secs_f64();
            state.metrics.observe_histogram("tuiyan_shizhang_miao", elapsed);
            state.metrics.inc_counter("tuiyan_cuo_wu_ji", 1);
            state.cb.record_failure();
            error!("inference failed for model={}: {}", req.model_id, e);
            (
                StatusCode::BAD_GATEWAY,
                Json(json!({"error": format!("inference failed: {}", e)})),
            )
                .into_response()
        }
    }
}

/// Cache statistics endpoint — returns a JSON snapshot of cache performance.
async fn cache_stats_handler(State(state): State<Arc<ChengTai>>) -> impl IntoResponse {
    let stats = state.cache.stats();
    (StatusCode::OK, Json(json!(stats))).into_response()
}

/// Cache eviction endpoint — evicts all cached entries belonging to a model.
///
/// Request body must contain JSON with a `model_id` field.
async fn cache_evict_handler(
    State(state): State<Arc<ChengTai>>,
    Json(body): Json<serde_json::Value>,
) -> impl IntoResponse {
    let model_id = match body.get("model_id").and_then(|v| v.as_str()) {
        Some(id) => id.to_string(),
        None => {
            return (
                StatusCode::BAD_REQUEST,
                Json(json!({"error": "missing 'model_id' field"})),
            )
                .into_response();
        }
    };

    match state.cache.evict_model(&model_id).await {
        Ok(count) => {
            info!("evicted {} cache entries for model={}", count, model_id);
            state
                .metrics
                .inc_counter("qingchu_zong_ji", count);
            (
                StatusCode::OK,
                Json(json!({"evicted": count, "model_id": model_id })),
            )
                .into_response()
        }
        Err(e) => {
            error!("cache evict failed for model={}: {}", model_id, e);
            (
                StatusCode::INTERNAL_SERVER_ERROR,
                Json(json!({"error": format!("cache evict failed: {}", e)})),
            )
                .into_response()
        }
    }
}

/// Build a permissive CORS layer suitable for development and internal
/// services.  In production, restrict `allow_origin` to known domains.
fn cors_layer() -> CorsLayer {
    CorsLayer::new()
        .allow_origin(Any)
        .allow_methods(Any)
        .allow_headers(Any)
}

/// ---------------------------------------------------------------------------
/// Router construction
/// ---------------------------------------------------------------------------

/// Build the Axum [`Router`] with all 非攻兼爱 REST 端点.
///
/// Layers are applied in reverse order of execution (bottom → top):
/// 1. `TraceLayer` — request/response logging.
/// 2. `CompressionLayer` — gzip / brotli response compression.
/// 3. `CorsLayer` — cross-origin header handling.
pub fn create_http_router(state: Arc<ChengTai>) -> Router {
    Router::new()
        .route("/health", get(health_handler))
        .route("/ready", get(ready_handler))
        .route("/metrics", get(metrics_handler))
        .route("/api/v1/infer", post(infer_handler))
        .route("/api/v1/cache/stats", get(cache_stats_handler))
        .route("/api/v1/cache/evict", post(cache_evict_handler))
        .route("/api/v1/dashboard", get(dashboard_handler))
        .route("/api/v1/alerts", get(alerts_handler))
        .layer(TraceLayer::new_for_http())
        .layer(CompressionLayer::new())
        .layer(cors_layer())
        .with_state(state)
}

/// Compute a deterministic xxHash of input tensors and return it as a
/// lowercase hex string for use as a cache key suffix.
fn compute_input_hash(inputs: &[Tensor]) -> String {
    let hash = hash_inputs(inputs);
    format!("{:016x}", hash)
}

/// 非攻兼爱主入口.
///
/// Orchestrates the full startup sequence:
/// 1. Parse `JuZiLing` from CLI arguments and environment variables.
/// 2. Initialise the `tracing` subscriber (JSON or pretty text).
/// 3. Create the `TianZhi`.
/// 4. Create the `FeiGong`.
/// 5. Build the tiered cache (L1 memory → L2 RocksDB → optional L3 S3).
/// 6. Connect to the model-loader service.
/// 7. Assemble the shared `ChengTai`.
/// 8. Bind the TCP listener and start the Axum HTTP server.
/// 9. Install signal handlers for graceful shutdown (`SIGTERM`, `SIGINT`).
/// 10. Serve with graceful shutdown and wait for completion.
#[tokio::main]
pub async fn main() -> Result<()> {
    // 1. Parse config from CLI / env
    let config = JuZiLing::parse();

    // 2. Initialize tracing subscriber
    let filter = EnvFilter::try_from_default_env()
        .unwrap_or_else(|_| EnvFilter::new(&config.log_level));
    if config.log_json {
        tracing_subscriber::fmt()
            .json()
            .with_env_filter(filter)
            .init();
    } else {
        tracing_subscriber::fmt().with_env_filter(filter).init();
    }

    info!("非攻兼爱系统启动 — 巨子令已授，四灵备战");
    info!("墨典配置: {:?}", config);

    // 3. Initialize metrics
    let metrics = Arc::new(TianZhi::new()?);

    // 4. Initialize circuit breaker
    let cb = Arc::new(FeiGong::new(
        config.cb_failure_threshold,
        config.cb_open_duration_secs,
        config.cb_half_open_successes,
    ));

    // 5. Initialize cache tiers
    let l1 = Arc::new(ZhuQue::new(config.l1_max_entries as u64, config.l1_ttl_secs));
    let l2 = Arc::new(XuanWu::new(&config.l2_path)?);
    let l3: Option<Arc<dyn CangFa>> = None; // Optional S3 tier
    let cache = Arc::new(JiGuanCheng::new(l1, l2, l3));

    // 6. Initialize loader client
    let loader = ZhiQi::new(config.loader_endpoint.clone())?;

    // 7. Build app state
    let app_state = Arc::new(ChengTai {
        config: config.clone(),
        metrics,
        cache,
        cb,
        loader,
    });

    // 8. Start HTTP server
    let addr: SocketAddr = config
        .http_addr
        .parse()
        .map_err(|e| FeiMingError::InvalidInput(format!("invalid http_addr: {}", e)))?;
    let listener = TcpListener::bind(addr).await.map_err(|e| {
        FeiMingError::Network(format!("failed to bind TCP listener on {}: {}", addr, e))
    })?;
    info!("HTTP server listening on {}", addr);

    let router = create_http_router(app_state.clone());

    // 9. Graceful shutdown — listen for SIGTERM and SIGINT
    let shutdown = CancellationToken::new();
    let shutdown_clone = shutdown.clone();
    tokio::spawn(async move {
        let mut sigterm = match signal(SignalKind::terminate()) {
            Ok(s) => s,
            Err(e) => {
                error!("failed to install SIGTERM handler: {}", e);
                return;
            }
        };
        let mut sigint = match signal(SignalKind::interrupt()) {
            Ok(s) => s,
            Err(e) => {
                error!("failed to install SIGINT handler: {}", e);
                return;
            }
        };
        tokio::select! {
            _ = sigterm.recv() => {
                info!("SIGTERM received — initiating graceful shutdown");
            }
            _ = sigint.recv() => {
                info!("SIGINT received — initiating graceful shutdown");
            }
        }
        shutdown_clone.cancel();
    });

    // 10. Serve with graceful shutdown
    axum::serve(listener, router.into_make_service())
        .with_graceful_shutdown(async move { shutdown.cancelled().await })
        .await
        .map_err(|e| FeiMingError::Network(format!("server error: {}", e)))?;

    info!("非攻兼爱系统已优雅关闭");
    Ok(())
}
